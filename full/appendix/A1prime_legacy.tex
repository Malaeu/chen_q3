\subsection{Local density with shifted atoms (legacy/appendix)}\label{app:legacy-a1-density}

This subsection records the shifted-atom proof that yields density on $[-R,R]$. It is
kept in the appendix because the detailed construction is not needed in the body text. The current
mainline theorem A1$'$ cites precisely this shifted density statement; what has been removed from the
main proof is only the old attempt to replace these shifted atoms by the centered specialization
$\tau=0$.

\begin{theorem}[A1' density with shifts (legacy)]
Let $K=[-R,R]$ with $R>0$. For $B>0$, $t>0$, $\tau\in[-R,R]$ define the \emph{normalized} even nonnegative frequency windows
\[
\Phi_{B,t,\tau}(\xi) := \tfrac{1}{2}\bigl[\Lambda_B(\xi-\tau)\, \rho_t(\xi-\tau) + \Lambda_B(\xi+\tau)\, \rho_t(\xi+\tau)\bigr],
\]
where $\Lambda_B(x)=(1-|x|/B)_+$ and $\rho_t(x)=(4\pi t)^{-1/2} e^{-x^2/(4t)}$ (so $\int_{\mathbb R}\rho_t=1$, $\rho_t\ge0$). The factor $\tfrac{1}{2}$ ensures that the symmetrization operator $\mathrm{Sym}(u):=(u(\cdot)+u(-\cdot))/2$ fixes even functions; scaling generators by a positive constant does not change the generated cone. Let $\mathcal C$ be the closed convex cone generated by finite nonnegative combinations of $\{\Phi_{B,t,\tau}\}$ with $\tau\in[-R,R]$ and $B$ sufficiently large (depending on $R$). Then $\mathcal C$ is dense in $C^+_{\mathrm{even}}([-R,R])$ in the uniform norm.
\end{theorem}
\begin{proof}
Fix $f\in C^+_{\mathrm{even}}([-R,R])$ and $\varepsilon>0$. Extend $f$ by zero to a compactly supported $\widetilde f\in C_c(\mathbb R)$ with $\widetilde f=f$ on $[-R,R]$.

Step 1 (mollification). Since $\rho_t$ is a positive approximate identity, there exists $t\in(0,t_0]$ such that
\begin{equation}\label{eq:A1_local_density-formula-6}
\sup_{|\xi|\le R} |(\widetilde f*\rho_t)(\xi)-f(\xi)| < \varepsilon/3.
\end{equation}
Set $g:=\widetilde f*\rho_t$. Then $g\ge0$, $g\in C^{\infty}(\mathbb R)$ and $g$ is even.

Step 2 (positive Riemann sums). Choose a uniform partition $-R=\tau_0<\tau_1<\dots<\tau_N=R$ with mesh $\Delta$ small enough so that
\begin{equation}\label{eq:A1_local_density-formula-5}
g_R(\xi):=\sum_{j=0}^{N-1} g(\tau_j^*)\,\rho_t(\xi-\tau_j^*)\,(\tau_{j+1}-\tau_j)
\end{equation}
satisfies $\sup_{|\xi|\le R} |g_R(\xi)-g(\xi)|<\varepsilon/3$ for some choices $\tau_j^*\in[\tau_j,\tau_{j+1}]$. Because the coefficients $g(\tau_j^*)(\tau_{j+1}-\tau_j)$ are nonnegative, $g_R$ is a finite nonnegative combination of translates of $\rho_t$.

Step 3 (Fej\'er truncation). For any $|\xi|,|\tau|\le R$ one has $|\Lambda_B(\xi-\tau)-1|\le (|\xi|+|\tau|)/B\le 2R/B$. Choosing $B\ge B_0:=6R/\varepsilon$ ensures
\begin{equation}\label{eq:A1_local_density-formula-4}
\sup_{|\xi|\le R,\,|\tau|\le R} |\Lambda_B(\xi-\tau)-1| < \varepsilon/3.
\end{equation}
Define the symmetric Fej\'er$\times$heat mixture as a sum of normalized atoms:
\begin{equation}\label{eq:A1_local_density-formula-3}
h(\xi):=\sum_{j=0}^{N-1} g(\tau_j^*) (\tau_{j+1}-\tau_j)\,\Phi_{B,t,\tau_j^*}(\xi).
\end{equation}
Since $g_R^{\mathrm{sym}}(\xi):=(g_R(\xi)+g_R(-\xi))/2$ is the symmetrized Riemann sum, and by~\eqref{eq:A1_local_density-formula-4} the Fej\'er factor satisfies $|\Lambda_B(\xi\pm\tau)-1|<\varepsilon/3$, we have for $|\xi|\le R$:
\begin{equation}\label{eq:A1_local_density-formula-2}
|h(\xi) - g_R^{\mathrm{sym}}(\xi)| \le (\varepsilon/3) \sum_j g(\tau_j^*)(\tau_{j+1}-\tau_j) \le C_f\,\varepsilon/3,
\end{equation}
with $C_f=\int_{-R}^{R} g(\tau)\,d\tau$ finite. Rescaling $\varepsilon$ by $3\max(1,C_f)$ if necessary, we get
\begin{equation}\label{eq:A1_local_density-formula-1}
\sup_{|\xi|\le R} |h(\xi) - g_R^{\mathrm{sym}}(\xi)| < \varepsilon/3.
\end{equation}

Step 4 (collect errors via triangle inequality). Since $g$ is even, $g_R^{\mathrm{sym}}$ satisfies $|g_R^{\mathrm{sym}}(\xi)-g(\xi)|<\varepsilon/3$ (by the Riemann sum bound plus symmetry contraction). Combined with Step~1 and Step~3:
\begin{equation}\label{eq:A1_local_density-formula}
\sup_{|\xi|\le R} |h(\xi)-f(\xi)| \le \sup|h-g_R^{\mathrm{sym}}| + \sup|g_R^{\mathrm{sym}}-g| + \sup|g-f| < \varepsilon/3+\varepsilon/3+\varepsilon/3 = \varepsilon.
\end{equation}
By construction, $h$ is a finite nonnegative combination of $\Phi_{B,t,\tau_j^*}$ with $\tau_j^*\in[-R,R]$, hence $h\in\mathrm{cone}\{\Phi_{B,t,\tau}\}$. Taking closures in $\|\cdot\|_\infty$ yields density of $\mathcal C$ in $C^+_{\mathrm{even}}([-R,R])$.
\end{proof}

\begin{lemma}[Fixed-$t_0$ cone density (legacy)]\label{lem:a1-fixed-t-density}
Fix $K>0$ and $t_0>0$. Let $\mathcal C_K(t_0)$ be the uniform-closure on $[-K,K]$
of the conoid generated by Fej\'er$\times$heat atoms $\{\Phi_{B,t_0,\tau}:\ B>0,\ |\tau|+B\le K\}$
and their even symmetrizations. Then $\mathcal C_K(t_0)=C^+_{\mathrm{even}}([-K,K])$.
\label{a1:lem:A1-fix-t}
\end{lemma}
\begin{proof}
Let $f\in C^+_{\mathrm{even}}([-K,K])$ and $\varepsilon>0$. Choose a uniform grid with mesh $\delta>0$
and hats $H_j(\xi):=\Lambda_\delta(\xi-\tau_j)$. By positive piecewise-linear interpolation,
$h(\xi):=\sum_j f(\tau_j)H_j(\xi)$ satisfies $\|h-f\|_\infty<\varepsilon/3$ and $h\ge0$.
Since $\rho_{t_0}$ is positive and Lipschitz on $[-2K,2K]$, there is $L_{t_0}>0$ with
$|\rho_{t_0}(x)-\rho_{t_0}(y)|\le L_{t_0}|x-y|$ for $x,y\in[-2K,2K]$. Set
$g(\xi):=\sum_j c_j\,\Phi_{\delta,t_0,\tau_j}(\xi)$ with $c_j:=f(\tau_j)/\rho_{t_0}(\tau_j)$.
Then for $|\xi|\le K$,
\[
g(\xi)-h(\xi)=\sum_j c_j\,\Lambda_\delta(\xi-\tau_j)\bigl(\rho_{t_0}(\xi-\tau_j)-\rho_{t_0}(\tau_j)\bigr),
\]
and therefore
\[
|g(\xi)-h(\xi)|\le L_{t_0}\,\delta\sum_j |c_j|\,\Lambda_\delta(\xi-\tau_j)\le L_{t_0}\,\delta\,\|c\|_{\ell^1}.
\]
Choosing $\delta$ small gives $\|g-f\|_\infty<\varepsilon$. Evenization preserves nonnegativity. Taking the conoid-closure
yields the claim.
\end{proof}
