\begin{lemma}[Closed-form upper bound for the prime trace]\label{pm:lem:rho-closed-form}
For $t>0$ one has
\begin{equation}\label{eq:Prime_trace_closed_form-formula-1}
  \rho(t)\ \le\ 2\int_0^{\infty} y\,e^{y/2}\,e^{-4\pi^2 t\,y^2}\,dy.
\end{equation}
With $a=4\pi^2 t$ and $b=\tfrac12$ this implies
\begin{equation}\label{eq:rho-closed-form}
  \rho(t)\ \le\ \frac{1}{4\pi^2 t}\ +\ \frac{\sqrt\pi}{2\,(4\pi^2 t)^{3/2}}\,\exp\!\Big(\,\frac{1}{16\pi^2 t}\,\Big).
\end{equation}
In particular, at the fixed scale $t=t_{\mathrm{crit}}$ this yields the cap $\|T_P^{\mathrm{RKHS}}(t)\|\le \rho(t)$ for all compacts.
\end{lemma}
\begin{proof}
The display \eqref{eq:Prime_trace_closed_form-formula-1} is Lemma~\ref{pm:lem:sum-to-rho}. Complete the square: $\int_0^{\infty} y\,e^{-a y^2 + b y}\,dy$ admits the identity
\(
  e^{\frac{b^2}{4a}}\,\frac{b\sqrt\pi}{4a^{3/2}}\,\bigl(1+\operatorname{erf}(\tfrac{b}{2\sqrt a})\bigr)\ +\ \frac{1}{2a}.
\)
Using $1+\operatorname{erf}(x)\le2$ gives the upper bound \eqref{eq:rho-closed-form}. Plug $a=4\pi^2 t$, $b=\tfrac12$ and simplify.
At $t=1$, $\pi>3$ implies $4\pi^2>36$ and hence $1/(4\pi^2)<1/36$. Moreover
\begin{align*}
\frac{\sqrt{\pi}}{2(4\pi^2)^{3/2}}\exp\!\Big(\frac{1}{16\pi^2}\Big)
&\le \frac{\sqrt{\pi}}{2\cdot 36^{3/2}}\exp\!\Big(\frac{1}{144}\Big)
\le \frac{2}{432}\cdot\exp\!\Big(\frac{1}{144}\Big) \\
&\le \frac{2}{432}\Big(1+\frac{1}{144}+\frac{1}{20736}\Big)
< \frac{2}{432}\cdot\frac{1007}{1000}
< \frac{1}{200},
\end{align*}
so the total bound is
\(
\tfrac{1}{36}+\tfrac{1}{200}
=\tfrac{59}{1800}
<\tfrac{72}{1800}
=\tfrac{1}{25}.
\)
\end{proof}

\begin{lemma}[Unshifted trace cap]\label{lem:shift-trace-cap}
Fix $K>0$. For any $B>0$ and $t>0$, the symmetrized prime sampling operator satisfies
\begin{equation}\label{eq:Prime_trace_closed_form-formula}
\|T_P^{\mathrm{RKHS}}(t)\|_{L^2\to L^2}\ \le\ \mathrm{tr}\,T_P^{\mathrm{RKHS}}(t)
= 2\sum_{n\ge2}\frac{\Lambda(n)}{\sqrt n}\,e^{-4\pi^2 t\,(\log n/(2\pi))^2}
\ \le\ \rho(t).
\end{equation}
\end{lemma}
\begin{proof}
Start with $\|T_P^{\mathrm{RKHS}}(t)\|\le \mathrm{tr}\,T_P^{\mathrm{RKHS}}(t)$ (PSD, finite rank on compacts).
Bound the sum by an integral of the positive integrand and apply the change $x=e^{y}$:
\begin{equation}\label{eq:Prime_trace_closed_form-formula-2}
\int_1^\infty \frac{\log x}{\sqrt x}\,e^{-4\pi^2 t(\log x)^2}\,dx
= \int_0^\infty y\,e^{y/2}e^{-4\pi^2 t\,y^2}\,dy.
\end{equation}
Doubling for $\pm\xi_n$ yields the stated bound by \eqref{eq:Prime_trace_closed_form-formula-1}.
\end{proof}
