\subsection{Main closure: from centered packets to Weil positivity}\label{sec:Main-closure}

\subsection*{Standing contract}
For each compact window $[-K,K]$ we distinguish:
\[
  R_K := C^+_{\mathrm{even}}([-K,K]),
  \qquad
  B_K := \{\Phi\in C_c(\mathbb R): \operatorname{supp}\Phi\subset[-K,K],\ \Phi\ \text{even}\},
\]
and the corrected local Weil-positive cone
\[
  \widetilde{\psi}(x):=\overline{\psi(-x)},
  \qquad
  \mathcal W_{K,0}^{\mathrm{pd}}
  := \bigl\{\psi * \widetilde{\psi}:\ \psi\in C_c^\infty(\mathbb R),\ \operatorname{supp}\psi\subset[-K/2,K/2]\bigr\},
  \qquad
  \mathcal W_K^{\mathrm{pd}}
  := \overline{\operatorname{cone}\!\bigl(\mathcal W_{K,0}^{\mathrm{pd}}\bigr)}^{\|\cdot\|_\infty},
  \qquad
  \mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}}
  := \operatorname{cone}\!\bigl\{\Phi_\Psi=\Psi*\widetilde{\Psi}:\ \Psi\in \mathcal P_K(t_0)\bigr\},
  \qquad
  \mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}
  := \operatorname{cone}\!\bigl\{\Phi_{B,t,p}=\Phi_{B,t}\,|p|^2:\ \Phi_{B,t}\ \text{centered Fej\'er$\times$heat},\ p\in\mathcal P_M\bigr\},
  \qquad
  \mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}
  \subseteq \mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}},
  \qquad
  \mathcal W^{\mathrm{pd}}:=\varinjlim_{K\to\infty}\mathcal W_K^{\mathrm{pd}}.
\]
The current draft already supplies:
\begin{itemize}
  \item \textbf{(T0)} the Guinand--Weil crosswalk, cf.\ Proposition~\ref{prop:T0-GW};
  \item \textbf{(A1$'$)} auxiliary density of shifted evenized Fej\'er$\times$heat atoms on the broad restriction cone $R_K$, cf.\ Theorem~\ref{a1:thm:A1-local-density};
  \item \textbf{(A2)} continuity of $Q^\star(t;\cdot)$ on the ambient compact-support class $B_K$, cf.\ Section~\ref{sec:A2};
  \item \textbf{(A3)+(RKHS)} the centered positivity mechanism at the critical scale.
\end{itemize}

What remains unresolved is the corrected density gate:
\begin{itemize}
  \item \textbf{(A1-pd)} Prove that the centered density family $\mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}}$ is dense in $\mathcal W_K^{\mathrm{pd}}$.
  \item \textbf{(packet-Rayleigh, background candidate)} The naive family $\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}=\operatorname{cone}\{\Phi_{B,t}|p|^2\}$ is mathematically natural, but it is too large to serve as the closure family.
  \item \textbf{(OP-pd)} Freeze an exact smaller operator-controlled family $\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}\subseteq \mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}}$ and identify $Q^\star(t;\Phi)$ with the centered Toeplitz/RKHS quadratic form on that same family.
  \item \textbf{(LF-pd)} Once local positivity on every $\mathcal W_K^{\mathrm{pd}}$ is available, lift it to the global cone $\mathcal W^{\mathrm{pd}}$.
\end{itemize}

\begin{lemma}[Naive packet-Rayleigh candidate]\label{lem:packet-rayleigh-identification}
For every compact window $[-K,K]$, every centered Fej\'er$\times$heat window $\Phi_{B,t}$, and every trigonometric polynomial $p\in\mathcal P_M$, define
\[
  \Phi_{B,t,p}(\xi):=\Phi_{B,t}(\xi)\,|p(\xi)|^2.
\]
Then
\[
  Q^\star(t;\Phi_{B,t,p})
  =
  2\pi\,
  \bigl\langle (T_M[P_A]-T_P^{\mathrm{Ray}}(t,M))\,p,\ p\bigr\rangle_{L^2(\mathbb T)}.
\]
Equivalently, positivity of the centered Toeplitz/RKHS quadratic form on the packet span yields
\[
  Q^\star(t;\Phi)\ge 0\qquad\text{for every }\Phi\in\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}.
\]
\end{lemma}

\begin{remark}[Why the naive candidate is too large]
The present draft already points toward Lemma~\ref{lem:packet-rayleigh-identification}:
Lemma~\ref{lem:a3-rayleigh-quotient} gives the quadratic Rayleigh pairing for arbitrary
trigonometric polynomials, while Theorem~\ref{thm:a3-rayleigh-identification} is the
special case $p\equiv1$ for a single centered Fej\'er$\times$heat window. What is
mathematically natural here is the family $\Phi_{B,t}|p|^2$. But on compact windows
$K<\pi$, a fixed centered window $\Phi_{B,t}$ with $B>K$ is strictly positive on
$[-K,K]$, and even real trigonometric polynomials are uniformly dense on that
interval. Hence $\Phi_{B,t}r^2$ is already dense in the broad local cone of even
nonnegative bumps. Combined with the full quadratic-form meaning of
Lemma~\ref{lem:packet-rayleigh-identification} and the centered A3 positivity
engine, this would force false broad local positivity. Therefore
$\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}$ is background-only, and the live task is
to construct a \emph{smaller} operator-controlled family
$\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}\subseteq\mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}}$.
\end{remark}

\begin{theorem}[Conditional main positivity on $\mathcal W^{\mathrm{pd}}$]\label{thm:Main-positivity}
Assume \textup{(T0)}, \textup{(A2)}, and Theorem~\ref{thm:A1-pd}. Assume further that for every compact window $[-K,K]$ there exists a centered operator-controlled family
\[
  \mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}
  \subseteq \mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}}
  \subseteq \mathcal W_K^{\mathrm{pd}}
\]
whose cone is dense in $\mathcal W_K^{\mathrm{pd}}$, and on which the centered Toeplitz/RKHS engine yields
\[
  Q^\star(t;\Phi)\ge0\ \text{ for every }\Phi\in\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}.
\]
Assume moreover that these local positivity statements are compatible
with the natural embeddings $\mathcal W_K^{\mathrm{pd}}\hookrightarrow \mathcal W_{K'}^{\mathrm{pd}}$.
Then
\[
  Q^\star(t;\Phi)\ \ge\ 0\qquad\text{for every }\Phi\in \mathcal W^{\mathrm{pd}},
\]
where $\mathcal W^{\mathrm{pd}}=\bigcup_{K>0}\mathcal W_K^{\mathrm{pd}}$ is the corrected Weil cone from Section~\ref{sec:scope}.
\end{theorem}

\begin{proof}
Fix $K>0$. By hypothesis, the cone generated by
$\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}$ is dense in $\mathcal W_K^{\mathrm{pd}}$ and
$Q^\star(t;\Phi)\ge0$ holds on that family. By Corollary~\ref{cor:A2-Lip}, $Q^\star(t;\cdot)$ is
continuous on the ambient compact-support class $B_K$, hence on
$\mathcal W_K^{\mathrm{pd}}\subseteq B_K$. Therefore the local positivity statement
extends to all of $\mathcal W_K^{\mathrm{pd}}$. Now let $\Phi\in\mathcal W^{\mathrm{pd}}$.
Choose $K$ with $\Phi\in\mathcal W_K^{\mathrm{pd}}$. Then $Q^\star(t;\Phi)\ge 0$, and
the compatibility assumption ensures that this conclusion is independent of the chosen compact exhaustion.
\end{proof}

\begin{remark}[Where the present draft stops]
The current analytic core supplies two separate ingredients:
\begin{enumerate}
  \item the centered positivity mechanism coming from A3 and the single-scale RKHS cap;
  \item the shifted density theorem A1$'$ on the broad restriction cone $R_K$.
\end{enumerate}
The target-cone audit shows that these do \emph{not} yet close RH. The public missing pair is now:
\begin{enumerate}
  \item the centered density theorem A1-pd on the corrected positive-definite cone $\mathcal W_K^{\mathrm{pd}}$;
  \item the exact operator-controlled packet theorem on a smaller family
  $\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}\subseteq \mathcal W_K^{\mathrm{pd}}$.
\end{enumerate}
\end{remark}

\begin{proposition}[Local closure on the corrected cone]\label{prop:compatibility-reduction-shifted-atoms}
Fix a compact window $[-K,K]$ and assume Theorem~\ref{thm:A1-pd} and
\[
  Q^\star(t;\Phi)\ge0\ \text{ for every }\Phi\in\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}},
\]
where $\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}\subseteq\mathcal W_K^{\mathrm{pd}}$
is a dense positive family.
Then
\[
  Q^\star(t;\Phi)\ge 0 \qquad \text{for every }\Phi\in \mathcal W_K^{\mathrm{pd}}.
\]
\end{proposition}

\begin{proof}
By the assumed packetwise inequality and linearity of $Q^\star$, the functional is
nonnegative on the cone generated by $\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}$.
By Corollary~\ref{cor:A2-Lip}, $Q^\star(t;\cdot)$ is continuous on the ambient class $B_K$ in the uniform norm, hence on $\mathcal W_K^{\mathrm{pd}}$.
Passing to the uniform limit along that dense positive family gives
$Q^\star(t;\Phi)\ge 0$ for every $\Phi\in\mathcal W_K^{\mathrm{pd}}$.
\end{proof}

\begin{remark}[Exact remaining mathematical task]\label{rem:compatibility-missing-scalar}
Proposition~\ref{prop:compatibility-reduction-shifted-atoms} shows that the corrected local closure step is routine once a dense positive family inside $\mathcal W_K^{\mathrm{pd}}$ has been identified. The unresolved work is therefore:
\begin{enumerate}
  \item prove the centered density theorem A1-pd on $\mathcal W_K^{\mathrm{pd}}$ via the density family $\mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}}$;
  \item keep the naive family $\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}$ background-only;
  \item freeze a smaller exact operator-controlled family
  $\mathcal G_{K,\mathrm{op}}^{\mathrm{pd}}\subseteq \mathcal G_{K,\mathrm{dens}}^{\mathrm{pd}}$
  together with the corresponding quadratic-form identity.
\end{enumerate}
Once that exact operator-family theorem is closed, the A3/RKHS positivity engine can feed the
corrected closure route without reopening the LF or Weil tail.
\end{remark}

\begin{remark}[Why shifted A1$'$ is now auxiliary]\label{rem:G1-frozen-target}
The shifted theorem A1$'$ remains mathematically valuable: it proves density on the broad restriction cone $R_K$ and it still supplies approximation technology. But after the target-cone audit it no longer serves as the public RH density theorem, because the corrected target is the positive-definite cone $\mathcal W_K^{\mathrm{pd}}$, not the full broad cone of even nonnegative compact-support tests.
\end{remark}
