\subsection{Main closure: from centered packets to Weil positivity}\label{sec:Main-closure}

\subsection*{Standing contract}
For each compact window $[-K,K]$ we distinguish:
\[
  R_K := C^+_{\mathrm{even}}([-K,K]),
  \qquad
  B_K := \{\Phi\in C_c(\mathbb R): \operatorname{supp}\Phi\subset[-K,K],\ \Phi\ \text{even}\},
\]
and the corrected local Weil-positive cone
\[
  \widetilde{\psi}(x):=\overline{\psi(-x)},
  \qquad
  \mathcal W_{K,0}^{\mathrm{pd}}
  := \bigl\{\psi * \widetilde{\psi}:\ \psi\in C_c^\infty(\mathbb R),\ \operatorname{supp}\psi\subset[-K/2,K/2]\bigr\},
  \qquad
  \mathcal W_K^{\mathrm{pd}}
  := \overline{\operatorname{cone}\!\bigl(\mathcal W_{K,0}^{\mathrm{pd}}\bigr)}^{\|\cdot\|_\infty},
  \qquad
  \mathcal G_K^{\mathrm{pd}}
  := \operatorname{cone}\!\bigl\{\Phi_\Psi=\Psi*\widetilde{\Psi}:\ \Psi\in \mathcal P_K(t_0)\bigr\},
  \qquad
  \mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}
  := \operatorname{cone}\!\bigl\{\Phi_{B,t,p}=\Phi_{B,t}\,|p|^2:\ \Phi_{B,t}\ \text{centered Fej\'er$\times$heat},\ p\in\mathcal P_M\bigr\},
  \qquad
  \Psi_c(x):=\sum_{j=-M}^M c_j\,g(x-j\Delta),
  \qquad
  h:=g*\widetilde g,
  \qquad
  \kappa_m:=Q^\star(t;h(\cdot-m\Delta)),
  \qquad
  S_{g,\Delta}(\theta):=\sum_{m\in\mathbb Z}\kappa_m e^{-im\theta},
  \qquad
  K_Q(g_i,g_j):=Q^\star(t;g_i * \widetilde{g_j}),
  \qquad
  \mathcal W^{\mathrm{pd}}:=\varinjlim_{K\to\infty}\mathcal W_K^{\mathrm{pd}}.
\]
The current draft already supplies:
\begin{itemize}
  \item \textbf{(T0)} the Guinand--Weil crosswalk, cf.\ Proposition~\ref{prop:T0-GW};
  \item \textbf{(A1$'$)} auxiliary density of shifted evenized Fej\'er$\times$heat atoms on the broad restriction cone $R_K$, cf.\ Theorem~\ref{a1:thm:A1-local-density};
  \item \textbf{(A2)} continuity of $Q^\star(t;\cdot)$ on the ambient compact-support class $B_K$, cf.\ Section~\ref{sec:A2};
  \item \textbf{(A3)+(RKHS)} the centered positivity mechanism at the critical scale.
\end{itemize}

What remains unresolved in the corrected route is now sharply localized:
\begin{itemize}
  \item \textbf{(A1-pd)} The centered density family $\mathcal G_K^{\mathrm{pd}}$ is the corrected local dense family.
  \item \textbf{(packet-Rayleigh-naive, background candidate)} The naive family $\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}=\operatorname{cone}\{\Phi_{B,t}|p|^2\}$ is mathematically natural, but it is too large to serve as the closure family.
  \item \textbf{(packet-Rayleigh-pd)} Exact Toeplitz quadratic-form identification on autocorrelation packets $\Psi_c * \widetilde{\Psi_c}$ is the honest bridge on the corrected cone.
  \item \textbf{(A3-pd, rejected theorem shape)} A uniform packet-symbol floor on the whole dense packet dictionary is too strong to be the live theorem.
  \item \textbf{(PSD-pd)} The remaining hard theorem is positive semidefiniteness of the packet kernel
  $K_Q(g_i,g_j)=Q^\star(t;g_i * \widetilde{g_j})$ on the same exact dense pre-packet space.
  \item \textbf{(LF-pd)} Once local positivity on every $\mathcal W_K^{\mathrm{pd}}$ is available, lift it to the global cone $\mathcal W^{\mathrm{pd}}$.
\end{itemize}

\begin{lemma}[Naive packet-Rayleigh candidate]\label{lem:packet-rayleigh-identification}
For every compact window $[-K,K]$, every centered Fej\'er$\times$heat window $\Phi_{B,t}$, and every trigonometric polynomial $p\in\mathcal P_M$, define
\[
  \Phi_{B,t,p}(\xi):=\Phi_{B,t}(\xi)\,|p(\xi)|^2.
\]
Then
\[
  Q^\star(t;\Phi_{B,t,p})
  =
  2\pi\,
  \bigl\langle (T_M[P_A]-T_P^{\mathrm{Ray}}(t,M))\,p,\ p\bigr\rangle_{L^2(\mathbb T)}.
\]
Equivalently, positivity of the centered Toeplitz/RKHS quadratic form on the packet span yields
\[
  Q^\star(t;\Phi)\ge 0\qquad\text{for every }\Phi\in\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}.
\]
\end{lemma}

\begin{remark}[Why the naive candidate is too large]
The present draft already points toward Lemma~\ref{lem:packet-rayleigh-identification}:
Lemma~\ref{lem:a3-rayleigh-quotient} gives the quadratic Rayleigh pairing for arbitrary
trigonometric polynomials, while Theorem~\ref{thm:a3-rayleigh-identification} is the
special case $p\equiv1$ for a single centered Fej\'er$\times$heat window. What is
mathematically natural here is the family $\Phi_{B,t}|p|^2$. But on compact windows
$K<\pi$, a fixed centered window $\Phi_{B,t}$ with $B>K$ is strictly positive on
$[-K,K]$, and even real trigonometric polynomials are uniformly dense on that
interval. Hence $\Phi_{B,t}r^2$ is already dense in the broad local cone of even
nonnegative bumps. Combined with the full quadratic-form meaning of
Lemma~\ref{lem:packet-rayleigh-identification} and the centered A3 positivity
engine, this would force false broad local positivity. Therefore
$\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}$ is background-only. The live task is not
to salvage that overlarge family, but to prove PSD of the packet kernel on the
exact dense autocorrelation packet family from A1-pd.
\end{remark}

\begin{lemma}[P1: exact packet sesquilinear identity]\label{lem:packet-sesquilinear}
Fix $K>0$ and let $\Psi,\Phi\in C_c([-K/2,K/2])$. Set
\[
  \widetilde{\Phi}(x):=\overline{\Phi(-x)},
  \qquad
  K_Q(\Psi,\Phi):=Q^\star(t;\Psi * \widetilde{\Phi}).
\]
Then
\[
  K_Q(\Psi,\Phi)
  =
  \iint_{\mathbb R^2} a^*(x-y)\Psi(x)\overline{\Phi(y)}\,dx\,dy
  -
  \sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}
  \int_{\mathbb R}\Psi(x)\overline{\Phi(x-\xi_n)}\,dx,
\]
where the prime sum is finite on the fixed compact window.
\end{lemma}

\begin{proof}
By definition,
\[
  (\Psi * \widetilde{\Phi})(\xi)
  =
  \int_{\mathbb R}\Psi(x)\overline{\Phi(x-\xi)}\,dx.
\]
Substituting this into the compact-window formula for $Q^\star(t;\cdot)$ and
interchanging integrals gives the Archimedean term. On the fixed compact only
finitely many prime nodes $\xi_n$ are active, so the prime term is finite and
the same substitution yields the displayed identity.
\end{proof}

\begin{theorem}[P2: Toeplitz reduction on a translation packet dictionary]\label{thm:packet-rayleigh-pd}
Let $g\in C_c(\mathbb R)$, let $\Delta>0$, and let
\[
  \Psi_c(x):=\sum_{j=-M}^M c_j\,g(x-j\Delta),
  \qquad
  h:=g*\widetilde g.
\]
Set
\[
  \kappa_m:=Q^\star(t;h(\cdot-m\Delta)),
  \qquad
  S_{g,\Delta}(\theta):=\sum_{m\in\mathbb Z}\kappa_m e^{-im\theta}.
\]
Then
\[
  Q^\star(t;\Psi_c * \widetilde{\Psi_c})
  =
  \sum_{i,j=-M}^M \kappa_{i-j} c_i\overline{c_j}
  =
  \bigl\langle T_M[S_{g,\Delta}]c,c\bigr\rangle.
\]
\end{theorem}

\begin{proof}
Expand
\[
  \Psi_c * \widetilde{\Psi_c}
  =
  \sum_{i,j=-M}^M c_i\overline{c_j}\,
  g(\cdot-i\Delta)*\widetilde{g(\cdot-j\Delta)}
  =
  \sum_{i,j=-M}^M c_i\overline{c_j}\,
  h(\cdot-(i-j)\Delta).
\]
Applying linearity of $Q^\star(t;\cdot)$ gives
\[
  Q^\star(t;\Psi_c * \widetilde{\Psi_c})
  =
  \sum_{i,j=-M}^M c_i\overline{c_j}\,\kappa_{i-j}.
\]
This is precisely the Toeplitz quadratic form associated with the symbol
$S_{g,\Delta}(\theta)=\sum_{m\in\mathbb Z}\kappa_m e^{-im\theta}$.
\end{proof}

\begin{proposition}[P3: desired PSD factorization of the packet prime block]\label{prop:desired-prime-factorization}
Fix a dense packet space $\mathcal P_K\subset C_c([-K/2,K/2])$ and define
\[
  K_P(\Psi,\Phi)
  :=
  \sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}\,(\Psi * \widetilde{\Phi})(\xi_n).
\]
If there exist a Hilbert space $\mathcal H_{P,K}$ and a map
$J_K:\mathcal P_K\to\mathcal H_{P,K}$ such that
\[
  K_P(\Psi,\Phi)=\langle J_K\Psi,J_K\Phi\rangle_{\mathcal H_{P,K}}
\]
for all $\Psi,\Phi\in\mathcal P_K$, then the packet prime block is positive
semidefinite on $\mathcal P_K$.
\end{proposition}

\begin{proof}
This is immediate from the Hilbert-space representation: every finite matrix
$[K_P(\Psi_i,\Psi_j)]$ is then a Gram matrix.
\end{proof}

\begin{lemma}[P5: decomposition of the full packet Toeplitz sequence]\label{lem:packet-symbol-decomposition}
Writing
\[
  \kappa_m
  =
  \int_{\mathbb R} a(\xi)\,h(\xi-m\Delta)\,d\xi
  -
  \sum_{n\ge2} w(n)\,h(\xi_n-m\Delta),
\]
the packet symbol splits as
\[
  S_{g,\Delta}(\theta)=A_{g,\Delta}(\theta)-P_{g,\Delta}(\theta),
\]
where
\[
  A_{g,\Delta}(\theta)
  :=
  \sum_{m\in\mathbb Z} e^{-im\theta}\!\int_{\mathbb R} a(\xi)\,h(\xi-m\Delta)\,d\xi,
  \qquad
  P_{g,\Delta}(\theta)
  :=
  \sum_{m\in\mathbb Z} e^{-im\theta}\!\sum_{n\ge2} w(n)\,h(\xi_n-m\Delta).
\]
Equivalently, for the twisted periodization
\[
  H_{g,\Delta,\theta}(\xi):=\sum_{m\in\mathbb Z} h(\xi-m\Delta)e^{-im\theta},
\]
one has
\[
  S_{g,\Delta}(\theta)
  =
  \int_{\mathbb R} a(\xi)\,H_{g,\Delta,\theta}(\xi)\,d\xi
  -
  \sum_{n\ge2} w(n)\,H_{g,\Delta,\theta}(\xi_n).
\]
Thus the symbol decomposition $S_{g,\Delta}=A_{g,\Delta}-P_{g,\Delta}$ remains a
useful structural object on the packet side. But on a dense packet dictionary it
is no longer honest to take one uniform lower bound for $S_{g,\Delta}$ as the
public theorem shape.
\end{lemma}

\begin{proof}
The identity $\kappa_m=\alpha_m-\beta_m$ is just the compact-window definition
of $Q^\star(t;\cdot)$ applied to $h(\cdot-m\Delta)$. Summing against
$e^{-im\theta}$ yields the displayed formulas for $S_{g,\Delta}$,
$A_{g,\Delta}$, and $P_{g,\Delta}$.
\end{proof}

\begin{proposition}[P4: prime-block obstruction on packet spaces]\label{prop:packet-prime-not-psd}
Fix a compact window $[-K,K]$ containing an active node $\xi_{n_0}>0$. Then on a
translation packet space supported in $[-K/2,K/2]$ the packet prime block
\[
  K_P(\Psi,\Phi)
  :=
  \sum_{n\ge2} w(n)\,(\Psi * \widetilde{\Phi})(\xi_n)
\]
is not positive semidefinite in general. In particular, no theorem of the form
\[
  K_P(\Psi,\Psi)=\|J\Psi\|_{\mathcal H}^2
\]
can hold on every dense packet dictionary containing such translates.
\end{proposition}

\begin{proof}
Since the active node set on $[-K,K]$ is finite and has positive minimum spacing,
we may choose a nonzero real packet $g$ with support so small that the translates
\[
  g,\qquad g(\cdot-\xi_{n_0}),\qquad g(\cdot-2\xi_{n_0})
\]
are pairwise disjoint and, moreover, for every other active node $\xi_n\neq\xi_{n_0}$
all overlaps contributing to $(g-g(\cdot-\xi_{n_0}))*\widetilde{(g-g(\cdot-\xi_{n_0}))}(\xi_n)$
vanish. Set
\[
  \Psi:=g-g(\cdot-\xi_{n_0}).
\]
Then
\[
  (\Psi * \widetilde{\Psi})(\xi_{n_0})
  =
  \langle \Psi, T_{\xi_{n_0}}\Psi\rangle
  =
  -\|g\|_{L^2}^2<0,
\]
while $(\Psi * \widetilde{\Psi})(\xi_n)=0$ for every other active node $\xi_n$.
Hence
\[
  K_P(\Psi,\Psi)
  =
  w(n_0)\,(\Psi * \widetilde{\Psi})(\xi_{n_0})
  <
  0.
\]
Therefore the packet prime block is not positive semidefinite on such packet spaces.
\end{proof}

\begin{proposition}[Why the uniform packet-symbol floor is too strong]\label{prop:a3-pd-too-strong}
Let $g\in C_c(\mathbb R)$ be nonzero and real-valued, and for $\Delta>0$ define
\[
  \Psi_\Delta:=g-g(\cdot-\Delta),
  \qquad
  \Phi_\Delta:=\Psi_\Delta * \widetilde{\Psi_\Delta}.
\]
Then $\Psi_\Delta\to0$ in $L^1$ as $\Delta\downarrow0$, hence
$\Phi_\Delta\to0$ uniformly on every fixed compact window. By the A2 continuity
module,
\[
  Q^\star(t;\Phi_\Delta)\to0.
\]
In particular, on any dense packet dictionary with arbitrarily fine translates,
no theorem of the form
\[
  Q^\star(t;\Psi * \widetilde{\Psi})\ge c_K\|c\|_2^2
  \qquad\text{with one uniform }c_K>0
\]
can hold on the whole dense packet family.
\end{proposition}

\begin{proof}
The translation action is continuous on $L^1$, so
$\|g-g(\cdot-\Delta)\|_{L^1}\to0$ as $\Delta\downarrow0$. For autocorrelations,
\[
  \|\Psi_\Delta * \widetilde{\Psi_\Delta}\|_\infty
  \le \|\Psi_\Delta\|_{L^1}^2,
\]
hence $\Phi_\Delta\to0$ uniformly on every fixed compact. Since
$Q^\star(t;\cdot)$ is continuous on the ambient compact-support class by A2,
the displayed convergence of $Q^\star(t;\Phi_\Delta)$ follows. Therefore a
single uniform positive gap cannot hold on a dense packet dictionary with
arbitrarily fine translates.
\end{proof}

\begin{proposition}[P6: Toeplitz--Herglotz criterion]\label{prop:packet-herglotz}
Fix $g$ and $\Delta$ and let $(\kappa_m)_{m\in\mathbb Z}$ be the corresponding
packet Toeplitz sequence. The following are equivalent:
\begin{enumerate}
  \item every finite Toeplitz block $[\kappa_{i-j}]_{|i|,|j|\le M}$ is positive semidefinite;
  \item $(\kappa_m)_{m\in\mathbb Z}$ is a positive-definite sequence on $\mathbb Z$;
  \item there exists a finite positive measure $\mu_{g,\Delta}$ on $\mathbb T$
  such that
  \[
    \kappa_m=\int_{\mathbb T} e^{-im\theta}\,d\mu_{g,\Delta}(\theta).
  \]
\end{enumerate}
If in addition $(\kappa_m)\in\ell^1(\mathbb Z)$, these are also equivalent to
pointwise nonnegativity of the continuous symbol
\[
  S_{g,\Delta}(\theta)=\sum_{m\in\mathbb Z}\kappa_m e^{-im\theta}.
\]
\end{proposition}

\begin{proof}
This is the classical Toeplitz--Herglotz equivalence for sequences on
$\mathbb Z$. In the $\ell^1$ regime the Fourier series for $S_{g,\Delta}$
converges uniformly, so positivity of the representing measure is equivalent to
pointwise nonnegativity of the symbol.
\end{proof}

\begin{proposition}[P7: full-symbol domination criterion]\label{prop:full-symbol-domination}
Fix an admissible packet dictionary generated by $g$ and $\Delta$. Assume that
the Toeplitz coefficients $\kappa_m=Q^\star(t;h(\cdot-m\Delta))$ define a genuine
circle symbol and that
\[
  A_{g,\Delta}(\theta)\ge P_{g,\Delta}(\theta)
  \qquad\text{for all }\theta\in[-\pi,\pi].
\]
Then
\[
  S_{g,\Delta}(\theta)=A_{g,\Delta}(\theta)-P_{g,\Delta}(\theta)\ge0
  \qquad\text{for all }\theta,
\]
so every finite Toeplitz section of $(\kappa_m)_{m\in\mathbb Z}$ is positive
semidefinite. Consequently the packet kernel is positive semidefinite on the
corresponding translation packet space.
\end{proposition}

\begin{proof}
The pointwise domination gives a nonnegative symbol $S_{g,\Delta}$ on the circle.
Hence the associated Toeplitz quadratic form satisfies
\[
  \bigl\langle T_M[S_{g,\Delta}]c,c\bigr\rangle \ge 0
  \qquad\text{for every }c\in\mathbb C^{2M+1}.
\]
By Theorem~\ref{thm:packet-rayleigh-pd} this quadratic form is exactly
$Q^\star(t;\Psi_c * \widetilde{\Psi_c})$, so the Toeplitz sections
$[\kappa_{i-j}]_{|i|,|j|\le M}$ are positive semidefinite. This is precisely the
packet-kernel positivity statement on that translation packet space.
\end{proof}

\begin{theorem}[P8: corrected compact positivity via the PSD packet kernel]\label{thm:PSD-pd}
Fix a compact window $[-K,K]$ and the dense pre-packet space
$\mathcal P_K(t_0)$ from Theorem~\ref{thm:A1-pd}. Assume that for every finite
packet family $\{g_i\}_{i=1}^N\subset \mathcal P_K(t_0)$ the matrix
\[
  \bigl[K_Q(g_i,g_j)\bigr]_{1\le i,j\le N}
  =
  \bigl[Q^\star(t;g_i * \widetilde{g_j})\bigr]_{1\le i,j\le N}
\]
is positive semidefinite. Then
\[
  Q^\star(t;\Phi)\ge0\qquad\text{for every }\Phi\in \mathcal G_K^{\mathrm{pd}}.
\]
\end{theorem}

\begin{proof}
Let $\Phi=\Psi * \widetilde{\Psi}\in \mathcal G_K^{\mathrm{pd}}$ with
\[
  \Psi=\sum_{i=1}^N c_i g_i,
  \qquad
  g_i\in\mathcal P_K(t_0).
\]
By bilinearity and linearity of $Q^\star(t;\cdot)$,
\[
  Q^\star(t;\Phi)
  =
  \sum_{i,j=1}^N c_i\overline{c_j}\,Q^\star(t;g_i * \widetilde{g_j})
  =
  \sum_{i,j=1}^N c_i\overline{c_j}\,K_Q(g_i,g_j).
\]
By the assumed positive semidefiniteness of the packet kernel matrix, this
quadratic form is nonnegative. Hence $Q^\star(t;\Phi)\ge0$ on the full dense
packet family.
\end{proof}

\begin{theorem}[Conditional main positivity on $\mathcal W^{\mathrm{pd}}$]\label{thm:Main-positivity}
Assume \textup{(T0)}, \textup{(A2)}, Theorem~\ref{thm:A1-pd},
Theorem~\ref{thm:packet-rayleigh-pd}, and Theorem~\ref{thm:PSD-pd}. Assume moreover
that these local positivity statements are compatible
with the natural embeddings $\mathcal W_K^{\mathrm{pd}}\hookrightarrow \mathcal W_{K'}^{\mathrm{pd}}$.
Then
\[
  Q^\star(t;\Phi)\ \ge\ 0\qquad\text{for every }\Phi\in \mathcal W^{\mathrm{pd}},
\]
where $\mathcal W^{\mathrm{pd}}=\bigcup_{K>0}\mathcal W_K^{\mathrm{pd}}$ is the corrected Weil cone from Section~\ref{sec:scope}.
\end{theorem}

\begin{proof}
Fix $K>0$. By Theorem~\ref{thm:A1-pd}, the cone generated by
$\mathcal G_K^{\mathrm{pd}}$ is dense in $\mathcal W_K^{\mathrm{pd}}$, and by
Theorem~\ref{thm:PSD-pd} we have $Q^\star(t;\Phi)\ge0$ on that same dense family.
By Corollary~\ref{cor:A2-Lip}, $Q^\star(t;\cdot)$ is
continuous on the ambient compact-support class $B_K$, hence on
$\mathcal W_K^{\mathrm{pd}}\subseteq B_K$. Therefore the local positivity statement
extends to all of $\mathcal W_K^{\mathrm{pd}}$. Now let $\Phi\in\mathcal W^{\mathrm{pd}}$.
Choose $K$ with $\Phi\in\mathcal W_K^{\mathrm{pd}}$. Then $Q^\star(t;\Phi)\ge 0$, and
the compatibility assumption ensures that this conclusion is independent of the chosen compact exhaustion.
\end{proof}

\begin{remark}[Where the present draft stops]
The current analytic core supplies two separate ingredients:
\begin{enumerate}
  \item the centered positivity mechanism coming from A3 and the single-scale RKHS cap;
  \item the shifted density theorem A1$'$ on the broad restriction cone $R_K$;
  \item the exact packet-Rayleigh theorem on autocorrelation packets.
\end{enumerate}
The target-cone audit shows that these do \emph{not} yet close RH. The public theorem package now contains the strict packet chain
\textup{(P1)}--\textup{(P6)}, but the public missing theorem is now:
\begin{enumerate}
  \item the PSD packet-kernel theorem PSD-pd on that same exact dense pre-packet space, with \textup{(P7)} as the clean sufficient criterion.
\end{enumerate}
\end{remark}

\begin{proposition}[Local closure on the corrected cone]\label{prop:compatibility-reduction-shifted-atoms}
Fix a compact window $[-K,K]$ and assume Theorem~\ref{thm:A1-pd} and
Theorem~\ref{thm:PSD-pd}.
Then
\[
  Q^\star(t;\Phi)\ge 0 \qquad \text{for every }\Phi\in \mathcal W_K^{\mathrm{pd}}.
\]
\end{proposition}

\begin{proof}
By the assumed packetwise inequality and linearity of $Q^\star$, the functional is
nonnegative on the cone generated by $\mathcal G_K^{\mathrm{pd}}$.
By Corollary~\ref{cor:A2-Lip}, $Q^\star(t;\cdot)$ is continuous on the ambient class $B_K$ in the uniform norm, hence on $\mathcal W_K^{\mathrm{pd}}$.
Passing to the uniform limit along that dense positive family gives
$Q^\star(t;\Phi)\ge 0$ for every $\Phi\in\mathcal W_K^{\mathrm{pd}}$.
\end{proof}

\begin{remark}[Exact remaining mathematical task]\label{rem:compatibility-missing-scalar}
Proposition~\ref{prop:compatibility-reduction-shifted-atoms} shows that the corrected local closure step is routine once a dense positive family inside $\mathcal W_K^{\mathrm{pd}}$ has been identified. The unresolved work is therefore:
\begin{enumerate}
  \item keep the naive family $\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}$ background-only;
  \item reject the old `A3-pd` uniform-gap route on the dense packet family;
  \item reject the literal `prime-block PSD factorization or Hilbert lift`
  theorem shape on packet space by Proposition~\ref{prop:packet-prime-not-psd};
  \item freeze the strict packet package
  \textup{(P1)} $\Rightarrow$ \textup{(P2)} $\Rightarrow$ \textup{(P4)}
  $\Rightarrow$ \textup{(P5)} $\Rightarrow$ \textup{(P6)};
  \item prove the PSD packet-kernel theorem PSD-pd for the same exact dense
  pre-packet space, with the symbol-side decomposition
  $S_{g,\Delta}=A_{g,\Delta}-P_{g,\Delta}$ kept as structural guidance and
  Proposition~\ref{prop:full-symbol-domination} (\textup{P7}) as the clean criterion.
\end{enumerate}
Once that packet-kernel theorem is closed, the A3/RKHS positivity engine can feed the
corrected closure route without reopening the LF or Weil tail.
\end{remark}

\begin{remark}[Two surviving strategy families for PSD-pd]\label{rem:psd-pd-strategies}
At the current state of the draft there are two honest ways to attack
Theorem~\ref{thm:PSD-pd}.
\begin{enumerate}
  \item \textbf{Herglotz/Bochner route (secondary diagnostic route).} Reinterpret
  the packet-kernel coefficients as Fourier data of a positive-definite sequence
  or measure, and deduce positive semidefiniteness of every finite packet-kernel
  matrix from that representation. This is the clean equivalence language for the
  target theorem, but by itself it does not yet provide the project-local
  constructive proof.
  \item \textbf{Direct full-kernel route (primary constructive route).}
  Keep the exact packet sesquilinear form on the dense pre-packet space, use
  Proposition~\ref{prop:packet-prime-not-psd} to rule out standalone PSD control
  of the prime block, and then attack positivity of the full kernel directly.
  The live theorem stack is:
  \textup{P1} exact packet sesquilinear identity
  $\Longrightarrow$
  \textup{P2} Toeplitz reduction
  $\Longrightarrow$
  \textup{P4} prime-block obstruction
  $\Longrightarrow$
  \textup{P5} full sequence split
  $\Longrightarrow$
  \textup{P6} Toeplitz--Herglotz criterion
  $\Longrightarrow$
  \textup{P7} full-symbol domination
  $A_{g,\Delta}\ge P_{g,\Delta}$ or a new operator package for the full kernel
  $\Longrightarrow$
  Theorem~\ref{thm:PSD-pd}.
\end{enumerate}
Neither route is yet written in the current draft, but these are now the exact
two strategy families that survive the rejection of the old uniform-gap theorem
shape.
\end{remark}

\begin{remark}[Why shifted A1$'$ is now auxiliary]\label{rem:G1-frozen-target}
The shifted theorem A1$'$ remains mathematically valuable: it proves density on the broad restriction cone $R_K$ and it still supplies approximation technology. But after the target-cone audit it no longer serves as the public RH density theorem, because the corrected target is the positive-definite cone $\mathcal W_K^{\mathrm{pd}}$, not the full broad cone of even nonnegative compact-support tests.
\end{remark}
