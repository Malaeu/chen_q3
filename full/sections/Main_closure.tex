\subsection{Main closure: from centered packets to Weil positivity}\label{sec:Main-closure}

\subsection*{Standing contract}
For each compact window $[-K,K]$ we distinguish:
\[
  R_K := C^+_{\mathrm{even}}([-K,K]),
  \qquad
  B_K := \{\Phi\in C_c(\mathbb R): \operatorname{supp}\Phi\subset[-K,K],\ \Phi\ \text{even}\},
\]
and the corrected local Weil-positive cone
\[
  \widetilde{\psi}(x):=\overline{\psi(-x)},
  \qquad
  \mathcal W_{K,0}^{\mathrm{pd}}
  := \bigl\{\psi * \widetilde{\psi}:\ \psi\in C_c^\infty(\mathbb R),\ \operatorname{supp}\psi\subset[-K/2,K/2]\bigr\},
  \qquad
  \mathcal W_K^{\mathrm{pd}}
  := \overline{\operatorname{cone}\!\bigl(\mathcal W_{K,0}^{\mathrm{pd}}\bigr)}^{\|\cdot\|_\infty},
  \qquad
  \mathcal G_K^{\mathrm{pd}}
  := \operatorname{cone}\!\bigl\{\Phi_\Psi=\Psi*\widetilde{\Psi}:\ \Psi\in \mathcal P_K(t_0)\bigr\},
  \qquad
  \mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}
  := \operatorname{cone}\!\bigl\{\Phi_{B,t,p}=\Phi_{B,t}\,|p|^2:\ \Phi_{B,t}\ \text{centered Fej\'er$\times$heat},\ p\in\mathcal P_M\bigr\},
  \qquad
  J\subset\mathbb Z\ \text{finite admissible dictionary},
  \qquad
  D(J):=\{i-j:\ i,j\in J\},
  \qquad
  \Psi_c(x):=\sum_{j\in J} c_j\,g(x-j\Delta),
  \qquad
  h:=g*\widetilde g,
  \qquad
  \kappa_m:=\mathcal Q(h(\cdot-m\Delta)),
  \qquad
  A_J(\theta):=\sum_{m\in D(J)}\alpha_m e^{-im\theta},
  \qquad
  P_J(\theta):=\sum_{m\in D(J)}\beta_m e^{-im\theta},
  \qquad
  S_J(\theta):=A_J(\theta)-P_J(\theta),
  \qquad
  S_{g,\Delta}(\theta):=\sum_{m\in\mathbb Z}\kappa_m e^{-im\theta},
  \qquad
  A_{J,r}:=P_r*A_J,
  \qquad
  P_{J,r}:=P_r*P_J,
  \qquad
  S_{J,r}:=P_r*S_J,
  \qquad
  K_Q(g_i,g_j):=\mathcal Q(g_i * \widetilde{g_j}),
  \qquad
  \mathcal W^{\mathrm{pd}}:=\varinjlim_{K\to\infty}\mathcal W_K^{\mathrm{pd}}.
\]
The current draft already supplies:
\begin{itemize}
  \item \textbf{(T0)} the Guinand--Weil crosswalk, cf.\ Proposition~\ref{prop:T0-GW};
  \item \textbf{(A1$'$)} auxiliary density of shifted evenized Fej\'er$\times$heat atoms on the broad restriction cone $R_K$, cf.\ Theorem~\ref{a1:thm:A1-local-density};
  \item \textbf{(A2)} continuity of $Q^\star(t;\cdot)$ on the ambient compact-support class $B_K$, cf.\ Section~\ref{sec:A2};
  \item \textbf{(A3)+(RKHS)} the centered positivity mechanism at the critical scale.
\end{itemize}

What remains unresolved in the corrected route is now sharply localized:
\begin{itemize}
  \item \textbf{(compact spectral route, diagnostic package)} The scalar spectral package
  \[
    \textup{S1 exact compact spectral identity}
    \Longrightarrow
    \textup{S2 spectral positivity criterion}
    \Longrightarrow
    \textup{S3 corrected compact positivity from }W_K(u)\ge0.
  \]
  This route attacks one scalar spectral weight per compact,
  \[
    W_K(u)=\widehat{a_K^*}(u)-\sum_{\xi_n\in\Xi_K}\frac{2\Lambda(n)}{\sqrt n}\cos(u\xi_n),
  \]
  and is therefore the primary constructive theorem shape.
  \item \textbf{(A1-pd)} The centered density family $\mathcal G_K^{\mathrm{pd}}$ is the corrected local dense family.
  \item \textbf{(packet-Rayleigh-naive, background candidate)} The naive family $\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}=\operatorname{cone}\{\Phi_{B,t}|p|^2\}$ is mathematically natural, but it is too large to serve as the closure family.
  \item \textbf{(packet-Rayleigh-pd)} Exact Toeplitz quadratic-form identification on autocorrelation packets $\Psi_c * \widetilde{\Psi_c}$ is the honest bridge on the corrected cone.
  \item \textbf{(A3-pd, rejected theorem shape)} A uniform packet-symbol floor on the whole dense packet dictionary is too strong to be the live theorem.
  \item \textbf{(PSD-pd)} The remaining hard theorem is positive semidefiniteness of the packet kernel
  $K_Q(g_i,g_j)=\mathcal Q(g_i * \widetilde{g_j})$ on a dense
  translation-compatible packet subspace feeding the corrected cone.
  \item \textbf{(finite P7, fallback discretization route)} For a fixed finite admissible packet dictionary, the fallback constructive target is the finite symbol
  $S_J(\theta)=A_J(\theta)-P_J(\theta)$ together with a Poisson-regularized
  verification step and explicit error budget. This route is now retained as a dictionary-level approximation / stress-test package for the scalar spectral route.
  \item \textbf{(bound package)} The immediate quantitative task is now a packet-level analogue of the old centered bridge:
  diagonal Archimedean core mass, off-diagonal Archimedean leakage, local prime
  collision mass, and the resulting explicit finite-dictionary inequalities
  \textup{(C1)} and \textup{(C1$'$)}.
  \item \textbf{(LF-pd)} Once local positivity on every $\mathcal W_K^{\mathrm{pd}}$ is available, lift it to the global cone $\mathcal W^{\mathrm{pd}}$.
\end{itemize}

\begin{lemma}[Naive packet-Rayleigh candidate]\label{lem:packet-rayleigh-identification}
For every compact window $[-K,K]$, every centered Fej\'er$\times$heat window $\Phi_{B,t}$, and every trigonometric polynomial $p\in\mathcal P_M$, define
\[
  \Phi_{B,t,p}(\xi):=\Phi_{B,t}(\xi)\,|p(\xi)|^2.
\]
Then
\[
  Q^\star(t;\Phi_{B,t,p})
  =
  2\pi\,
  \bigl\langle (T_M[P_A]-T_P^{\mathrm{Ray}}(t,M))\,p,\ p\bigr\rangle_{L^2(\mathbb T)}.
\]
Equivalently, positivity of the centered Toeplitz/RKHS quadratic form on the packet span yields
\[
  Q^\star(t;\Phi)\ge 0\qquad\text{for every }\Phi\in\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}.
\]
\end{lemma}

\begin{remark}[Why the naive candidate is too large]
The present draft already points toward Lemma~\ref{lem:packet-rayleigh-identification}:
Lemma~\ref{lem:a3-rayleigh-quotient} gives the quadratic Rayleigh pairing for arbitrary
trigonometric polynomials, while Theorem~\ref{thm:a3-rayleigh-identification} is the
special case $p\equiv1$ for a single centered Fej\'er$\times$heat window. What is
mathematically natural here is the family $\Phi_{B,t}|p|^2$. But on compact windows
$K<\pi$, a fixed centered window $\Phi_{B,t}$ with $B>K$ is strictly positive on
$[-K,K]$, and even real trigonometric polynomials are uniformly dense on that
interval. Hence $\Phi_{B,t}r^2$ is already dense in the broad local cone of even
nonnegative bumps. Combined with the full quadratic-form meaning of
Lemma~\ref{lem:packet-rayleigh-identification} and the centered A3 positivity
engine, this would force false broad local positivity. Therefore
$\mathcal G_{K,\mathrm{Ray}}^{\mathrm{pd}}$ is background-only. The live task is not
to salvage that overlarge family, but to prove PSD of the packet kernel on the
exact dense autocorrelation packet family from A1-pd.
\end{remark}

\begin{definition}[Symmetric packet extension of the compact Weil functional]\label{def:Q-symmetric-extension}
For a compactly supported test $F\in C_c(\mathbb R)$, define
\[
  \mathcal Q(F)
  :=
  \int_{\mathbb R} a^*(\xi)\,F(\xi)\,d\xi
  -
  \frac12\sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}\bigl(F(\xi_n)+F(-\xi_n)\bigr).
\]
If $F$ is even, then $\mathcal Q(F)=Q^\star(t;F)$.
\end{definition}

\begin{lemma}[P1: exact packet sesquilinear identity]\label{lem:packet-sesquilinear}
Fix $K>0$ and let $\Psi,\Phi\in C_c([-K/2,K/2])$. Set
\[
  \widetilde{\Phi}(x):=\overline{\Phi(-x)},
  \qquad
  K_Q(\Psi,\Phi):=\mathcal Q(\Psi * \widetilde{\Phi}).
\]
Then
\[
  K_Q(\Psi,\Phi)
  =
  \iint_{\mathbb R^2} a^*(x-y)\Psi(x)\overline{\Phi(y)}\,dx\,dy
  -
  \frac12\sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}
  \Bigl(
    \langle \Psi,T_{\xi_n}\Phi\rangle
    +
    \langle \Psi,T_{-\xi_n}\Phi\rangle
  \Bigr),
\]
where the prime sum is finite on the fixed compact window.
\end{lemma}

\begin{proof}
By definition,
\[
  (\Psi * \widetilde{\Phi})(\xi)
  =
  \int_{\mathbb R}\Psi(x)\overline{\Phi(x-\xi)}\,dx.
\]
Substituting this into the compact-window formula for $\mathcal Q(\cdot)$ and
interchanging integrals gives the Archimedean term. On the fixed compact only
finitely many prime nodes $\pm\xi_n$ are active, so the prime term is finite
and the same substitution yields the displayed identity.
\end{proof}

\begin{theorem}[P2: Toeplitz reduction on a finite translation packet dictionary]\label{thm:packet-rayleigh-pd}
Let $g\in C_c(\mathbb R)$, let $\Delta>0$, let $J\subset\mathbb Z$ be finite, and let
\[
  g_j(x):=g(x-j\Delta),
  \qquad
  \Psi_c(x):=\sum_{j\in J} c_j\,g_j(x),
  \qquad
  h:=g*\widetilde g,
  \qquad
  D(J):=\{i-j:\ i,j\in J\}.
\]
Set
\[
  \kappa_m:=\mathcal Q(h(\cdot-m\Delta))
  \qquad (m\in D(J)).
\]
Then
\[
  \mathcal Q(\Psi_c * \widetilde{\Psi_c})
  =
  \sum_{i,j\in J} \kappa_{i-j} c_i\overline{c_j}.
\]
\end{theorem}

\begin{proof}
Expand
\[
  \Psi_c * \widetilde{\Psi_c}
  =
  \sum_{i,j\in J} c_i\overline{c_j}\,
  g_i*\widetilde{g_j}
  =
  \sum_{i,j\in J} c_i\overline{c_j}\,
  h(\cdot-(i-j)\Delta).
\]
Applying linearity of $\mathcal Q(\cdot)$ gives
\[
  \mathcal Q(\Psi_c * \widetilde{\Psi_c})
  =
  \sum_{i,j\in J} c_i\overline{c_j}\,\kappa_{i-j}.
\]
\end{proof}

\begin{proposition}[P3: desired PSD factorization of the packet prime block]\label{prop:desired-prime-factorization}
Fix a dense packet space $\mathcal P_K\subset C_c([-K/2,K/2])$ and define
\[
  K_P(\Psi,\Phi)
  :=
  \frac12\sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}
  \bigl((\Psi * \widetilde{\Phi})(\xi_n)+(\Psi * \widetilde{\Phi})(-\xi_n)\bigr).
\]
If there exist a Hilbert space $\mathcal H_{P,K}$ and a map
$J_K:\mathcal P_K\to\mathcal H_{P,K}$ such that
\[
  K_P(\Psi,\Phi)=\langle J_K\Psi,J_K\Phi\rangle_{\mathcal H_{P,K}}
\]
for all $\Psi,\Phi\in\mathcal P_K$, then the packet prime block is positive
semidefinite on $\mathcal P_K$.
\end{proposition}

\begin{proof}
This is immediate from the Hilbert-space representation: every finite matrix
$[K_P(\Psi_i,\Psi_j)]$ is then a Gram matrix.
\end{proof}

\begin{lemma}[P5: decomposition of the full packet Toeplitz sequence]\label{lem:packet-symbol-decomposition}
Writing
\[
  \alpha_m:=\int_{\mathbb R} a^*(\xi)\,h(\xi-m\Delta)\,d\xi,
  \qquad
  \beta_m:=\frac12\sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}
  \Bigl(h(\xi_n-m\Delta)+h(\xi_n+m\Delta)\Bigr),
\]
one has $\kappa_m=\alpha_m-\beta_m$, and
the packet symbol splits as
\[
  S_{g,\Delta}(\theta)=A_{g,\Delta}(\theta)-P_{g,\Delta}(\theta),
\]
where
\[
  A_{g,\Delta}(\theta)
  :=
  \sum_{m\in\mathbb Z} \alpha_m e^{-im\theta},
  \qquad
  P_{g,\Delta}(\theta)
  :=
  \sum_{m\in\mathbb Z} \beta_m e^{-im\theta}.
\]
Equivalently, for the twisted periodization
\[
  H_{g,\Delta,\theta}(\xi):=\sum_{m\in\mathbb Z} h(\xi-m\Delta)e^{-im\theta},
\]
one has
\[
  S_{g,\Delta}(\theta)
  =
  \int_{\mathbb R} a^*(\xi)\,H_{g,\Delta,\theta}(\xi)\,d\xi
  -
  \frac12\sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}
  \Bigl(H_{g,\Delta,\theta}(\xi_n)+H_{g,\Delta,\theta}(-\xi_n)\Bigr).
\]
Thus the symbol decomposition $S_{g,\Delta}=A_{g,\Delta}-P_{g,\Delta}$ remains a
useful structural object on the packet side. But on a dense packet dictionary it
is no longer honest to take one uniform lower bound for $S_{g,\Delta}$ as the
public theorem shape.
\end{lemma}

\begin{proof}
The identity $\kappa_m=\alpha_m-\beta_m$ is just the compact-window definition
of $\mathcal Q(\cdot)$ applied to $h(\cdot-m\Delta)$. Summing against
$e^{-im\theta}$ yields the displayed formulas for $S_{g,\Delta}$,
$A_{g,\Delta}$, and $P_{g,\Delta}$.
\end{proof}

\begin{proposition}[P4: prime-block obstruction on packet spaces]\label{prop:packet-prime-not-psd}
Fix a compact window $[-K,K]$ containing an active node $\xi_{n_0}>0$. Then on a
translation packet space supported in $[-K/2,K/2]$ the packet prime block
\[
  K_P(\Psi,\Phi)
  :=
  \frac12\sum_{n\ge2}\frac{2\Lambda(n)}{\sqrt n}
  \bigl((\Psi * \widetilde{\Phi})(\xi_n)+(\Psi * \widetilde{\Phi})(-\xi_n)\bigr)
\]
is not positive semidefinite in general. In particular, no theorem of the form
\[
  K_P(\Psi,\Psi)=\|J\Psi\|_{\mathcal H}^2
\]
can hold on every dense packet dictionary containing such translates.
\end{proposition}

\begin{proof}
Since the active node set on $[-K,K]$ is finite and has positive minimum spacing,
we may choose a nonzero real packet $g$ with support so small that the translates
\[
  g,\qquad g(\cdot\pm\xi_{n_0}),\qquad g(\cdot-2\xi_{n_0})
\]
are pairwise disjoint and, moreover, for every other active node $\xi_n\neq\xi_{n_0}$
all overlaps contributing to
$(g-g(\cdot-\xi_{n_0}))*\widetilde{(g-g(\cdot-\xi_{n_0}))}(\pm\xi_n)$
vanish. Set
\[
  \Psi:=g-g(\cdot-\xi_{n_0}).
\]
Then
\[
  (\Psi * \widetilde{\Psi})(\xi_{n_0})
  =
  \langle \Psi, T_{\xi_{n_0}}\Psi\rangle
  =
  -\|g\|_{L^2}^2<0,
\]
and likewise
\[
  (\Psi * \widetilde{\Psi})(-\xi_{n_0})
  =
  \langle \Psi, T_{-\xi_{n_0}}\Psi\rangle
  =
  -\|g\|_{L^2}^2<0.
\]
For every other active node $\xi_n\neq\xi_{n_0}$, the chosen support separation
gives
\[
  (\Psi * \widetilde{\Psi})(\pm\xi_n)=0.
\]
Hence
\[
  K_P(\Psi,\Psi)
  =
  \frac12\frac{2\Lambda(n_0)}{\sqrt{n_0}}
  \Bigl(
    (\Psi * \widetilde{\Psi})(\xi_{n_0})
    +
    (\Psi * \widetilde{\Psi})(-\xi_{n_0})
  \Bigr)
  <
  0.
\]
Therefore the packet prime block is not positive semidefinite on such packet spaces.
\end{proof}

\begin{proposition}[Sparse Gershgorin criterion for packet Toeplitz blocks]\label{prop:packet-gershgorin}
Assume that the base packet $g$ is real and nonnegative, so that
$h=g*\widetilde g\ge0$ and hence $\beta_m\ge0$ for all $m$. Fix $M\ge1$. If
\[
  \kappa_0\ge \sum_{1\le |m|\le 2M} |\kappa_m|+\varepsilon
  \qquad\text{for some }\varepsilon>0,
\]
then the Toeplitz block $[\kappa_{i-j}]_{|i|,|j|\le M}$ is positive semidefinite,
indeed has least eigenvalue at least $\varepsilon$. In particular, it is enough
to prove the stronger split inequality
\[
  \alpha_0
  \ge
  \beta_0+\sum_{1\le |m|\le 2M}\bigl(|\alpha_m|+\beta_m\bigr)+\varepsilon.
\]
\end{proposition}

\begin{proof}
The Toeplitz block has diagonal entry $\kappa_0$ and off-diagonal entries
$\kappa_{i-j}$. The displayed diagonal-dominance hypothesis is exactly the
strict Gershgorin criterion, so every Gershgorin disc lies in
$[\varepsilon,\infty)$ and the block is positive semidefinite. Since
$\kappa_m=\alpha_m-\beta_m$ and $\beta_m\ge0$, the stronger split inequality
implies the diagonal-dominance bound immediately.
\end{proof}

\begin{proposition}[Why Gershgorin cannot be the dense main theorem]\label{prop:a3-pd-too-strong}
Let $g\in C_c(\mathbb R)$ be nonzero and real-valued, and for $\Delta>0$ define
\[
  \Psi_\Delta:=g-g(\cdot-\Delta),
  \qquad
  \Phi_\Delta:=\Psi_\Delta * \widetilde{\Psi_\Delta}.
\]
Then $\Psi_\Delta\to0$ in $L^1$ as $\Delta\downarrow0$, hence
$\Phi_\Delta\to0$ uniformly on every fixed compact window. By the A2 continuity
module,
\[
  Q^\star(t;\Phi_\Delta)\to0.
\]
In particular, on any dense packet dictionary with arbitrarily fine translates,
no theorem of the form
\[
  Q^\star(t;\Psi * \widetilde{\Psi})\ge c_K\|c\|_2^2
  \qquad\text{with one uniform }c_K>0
\]
can hold on the whole dense packet family.
\end{proposition}

\begin{proof}
The translation action is continuous on $L^1$, so
$\|g-g(\cdot-\Delta)\|_{L^1}\to0$ as $\Delta\downarrow0$. For autocorrelations,
\[
  \|\Psi_\Delta * \widetilde{\Psi_\Delta}\|_\infty
  \le \|\Psi_\Delta\|_{L^1}^2,
\]
hence $\Phi_\Delta\to0$ uniformly on every fixed compact. Since
$Q^\star(t;\cdot)$ is continuous on the ambient compact-support class by A2,
the displayed convergence of $Q^\star(t;\Phi_\Delta)$ follows. Therefore a
single uniform positive gap cannot hold on a dense packet dictionary with
arbitrarily fine translates.
\end{proof}

\begin{proposition}[P6: Toeplitz--Herglotz criterion]\label{prop:packet-herglotz}
Fix $g$ and $\Delta$ and let $(\kappa_m)_{m\in\mathbb Z}$ be the corresponding
packet Toeplitz sequence. The following are equivalent:
\begin{enumerate}
  \item every finite Toeplitz block $[\kappa_{i-j}]_{|i|,|j|\le M}$ is positive semidefinite;
  \item $(\kappa_m)_{m\in\mathbb Z}$ is a positive-definite sequence on $\mathbb Z$;
  \item there exists a finite positive measure $\mu_{g,\Delta}$ on $\mathbb T$
  such that
  \[
    \kappa_m=\int_{\mathbb T} e^{-im\theta}\,d\mu_{g,\Delta}(\theta).
  \]
\end{enumerate}
If in addition $(\kappa_m)\in\ell^1(\mathbb Z)$, these are also equivalent to
pointwise nonnegativity of the continuous symbol
\[
  S_{g,\Delta}(\theta)=\sum_{m\in\mathbb Z}\kappa_m e^{-im\theta}.
\]
\end{proposition}

\begin{proof}
This is the classical Toeplitz--Herglotz equivalence for sequences on
$\mathbb Z$. In the $\ell^1$ regime the Fourier series for $S_{g,\Delta}$
converges uniformly, so positivity of the representing measure is equivalent to
pointwise nonnegativity of the symbol.
\end{proof}

\begin{proposition}[P7.3: finite symbol identity]\label{prop:finite-symbol-identity}
Fix a finite admissible packet dictionary generated by $g$, $\Delta$, and
$J\subset\mathbb Z$. Define
\[
  A_J(\theta):=\sum_{m\in D(J)} \alpha_m e^{-im\theta},
  \qquad
  P_J(\theta):=\sum_{m\in D(J)} \beta_m e^{-im\theta},
\]
\[
  S_J(\theta):=A_J(\theta)-P_J(\theta)
  =
  \sum_{m\in D(J)} \kappa_m e^{-im\theta},
  \qquad
  p_c(\theta):=\sum_{j\in J} c_j e^{ij\theta}.
\]
Then
\[
  \mathcal Q(\Psi_c*\widetilde{\Psi_c})\ge0
  \quad\Longleftrightarrow\quad
  \frac{1}{2\pi}\int_{-\pi}^{\pi} S_J(\theta)\,|p_c(\theta)|^2\,d\theta\ge0,
\]
and in fact one has the exact identity
\[
  \mathcal Q(\Psi_c*\widetilde{\Psi_c})
  =
  \frac{1}{2\pi}\int_{-\pi}^{\pi} S_J(\theta)\,|p_c(\theta)|^2\,d\theta.
\]
\end{proposition}

\begin{proof}
By Theorem~\ref{thm:packet-rayleigh-pd},
\[
  \mathcal Q(\Psi_c * \widetilde{\Psi_c})
  =
  \sum_{m\in D(J)} \kappa_m
  \sum_{\substack{i,j\in J\\ i-j=m}} c_i\overline{c_j}.
\]
On the other hand,
\[
  |p_c(\theta)|^2
  =
  \sum_{i,j\in J} c_i\overline{c_j}e^{i(i-j)\theta}.
\]
Fourier orthogonality on $\mathbb T$ therefore yields the displayed identity.
\end{proof}

\begin{corollary}[P7.4: exact finite-symbol domination criterion]\label{cor:finite-symbol-domination}
Fix a finite admissible packet dictionary generated by $g$, $\Delta$, and
$J\subset\mathbb Z$. If
\[
  S_J(\theta)\ge0
  \qquad\text{for all }\theta\in[-\pi,\pi],
\]
equivalently
\[
  A_J(\theta)\ge P_J(\theta)
  \qquad\text{for all }\theta\in[-\pi,\pi],
\]
then
\[
  \mathcal Q(\Psi_c*\widetilde{\Psi_c})\ge0
  \qquad\text{for all }c\in\mathbb C^J.
\]
Hence the Toeplitz block
\[
  T_J(\kappa):=[\kappa_{i-j}]_{i,j\in J}
\]
is positive semidefinite.
\end{corollary}

\begin{proof}
This is immediate from Proposition~\ref{prop:finite-symbol-identity}, since
$|p_c(\theta)|^2\ge0$ pointwise on $\mathbb T$.
\end{proof}

\begin{proposition}[P7.5: Poisson-regularized verification criterion]\label{prop:poisson-regularized-p7}
Fix a finite admissible packet dictionary generated by $g$, $\Delta$, and
$J\subset\mathbb Z$. For $0<r<1$, let $P_r$ be the Poisson kernel on $\mathbb T$
and define
\[
  A_{J,r}:=P_r*A_J
  =
  \sum_{m\in D(J)} r^{|m|}\alpha_m e^{-im\theta},
\]
\[
  P_{J,r}:=P_r*P_J
  =
  \sum_{m\in D(J)} r^{|m|}\beta_m e^{-im\theta},
  \qquad
  S_{J,r}:=A_{J,r}-P_{J,r}=P_r*S_J.
\]
Then
\[
  \|S_J-S_{J,r}\|_\infty
  \le
  (1-r)\sum_{m\in D(J)} |m|\,|\kappa_m|
  \le
  (1-r)\sum_{m\in D(J)} |m|\bigl(|\alpha_m|+\beta_m\bigr).
\]
\end{proposition}

\begin{proof}
Since
\[
  S_J(\theta)-S_{J,r}(\theta)
  =
  \sum_{m\in D(J)}(1-r^{|m|})\kappa_m e^{-im\theta},
\]
we use
\[
  1-r^{|m|}\le (1-r)|m|
\]
and then take sup norms. The second inequality is the triangle inequality
together with $\kappa_m=\alpha_m-\beta_m$.
\end{proof}

\begin{corollary}[P7.6: regularized domination with explicit error budget]\label{cor:regularized-p7}
Fix a finite admissible packet dictionary generated by $g$, $\Delta$, and
$J\subset\mathbb Z$. Assume there exist $0<r<1$ and $\varepsilon_J>0$ such that
\[
  A_{J,r}(\theta)-P_{J,r}(\theta)\ge \varepsilon_J
  \qquad\text{for all }\theta\in[-\pi,\pi],
\]
and
\[
  (1-r)\sum_{m\in D(J)} |m|\bigl(|\alpha_m|+\beta_m\bigr)\le \varepsilon_J.
\]
Then
\[
  S_J(\theta)\ge0
  \qquad\text{for all }\theta\in[-\pi,\pi],
\]
hence
\[
  \mathcal Q(\Psi_c*\widetilde{\Psi_c})\ge0
  \qquad\text{for all }c\in\mathbb C^J.
\]
\end{corollary}

\begin{proof}
By Proposition~\ref{prop:poisson-regularized-p7},
\[
  S_J(\theta)
  \ge
  S_{J,r}(\theta)-\|S_J-S_{J,r}\|_\infty
  \ge
  \varepsilon_J-\varepsilon_J
  =0.
\]
The final conclusion follows from Corollary~\ref{cor:finite-symbol-domination}.
\end{proof}

\begin{definition}[Packet geometry for explicit coefficient bounds]\label{def:packet-bound-geometry}
Fix a compact window $[-K,K]$, a base packet $g\in C_c(\mathbb R)$, and an
admissible translation step $\Delta>0$. Write
\[
  \operatorname{supp} g\subset[-R_g,R_g],
  \qquad
  h:=g*\widetilde g,
  \qquad
  R_h:=2R_g,
\]
so that $\operatorname{supp} h\subset[-R_h,R_h]$. For $m\in\mathbb Z$ define
\[
  I_m:=[m\Delta-R_h,\ m\Delta+R_h].
\]
For $r\in(0,R_h]$ set
\[
  H_r:=\int_{|u|\le r} h(u)\,du,
  \qquad
  T_r:=\|h\|_{L^1}-H_r,
  \qquad
  m_r^*:=\inf_{|x|\le r} a^*(x),
  \qquad
  M_{r,K}^*:=\sup_{r<|x|\le R_h}|a^*(x)|.
\]
Let
\[
  \Xi_K:=\{\xi_n:\ 0<\xi_n\le K\},
  \qquad
  \delta_K:=\min_{\substack{\xi_m,\xi_n\in\Xi_K\\ \xi_m\neq \xi_n}}
  |\xi_m-\xi_n|,
  \qquad
  w_{\max}(K):=\max_{\xi_n\in\Xi_K}\frac{2\Lambda(n)}{\sqrt n}.
\]
Finally, define the local symmetric prime mass
\[
  \Pi_K^{\mathrm{sym}}(s;R)
  :=
  \frac12\sum_{\xi_n\in\Xi_K}\frac{2\Lambda(n)}{\sqrt n}
  \Bigl(
    \mathbf 1_{|\xi_n-s|\le R}
    +
    \mathbf 1_{|\xi_n+s|\le R}
  \Bigr),
  \qquad s\ge0.
\]
\end{definition}

\begin{lemma}[A1: support-localization bound for the Archimedean coefficients]\label{lem:packet-alpha-support-bound}
For every $m\in D(J)$ one has
\[
  |\alpha_m|
  \le
  \|a^*\|_{L^\infty(I_m)}\,\|h\|_{L^1}.
\]
\end{lemma}

\begin{proof}
Since $h$ is supported in $[-R_h,R_h]$,
\[
  \alpha_m
  =
  \int_{-R_h}^{R_h} a^*(m\Delta+u)\,h(u)\,du.
\]
Taking the supremum of $|a^*|$ on $I_m$ yields the claim.
\end{proof}

\begin{lemma}[A2: approximate-identity refinement for the Archimedean coefficients]\label{lem:packet-alpha-modulus-bound}
Assume $I_m\subset[-K,K]$. Then
\[
  |\alpha_m-a^*(m\Delta)\|h\|_{L^1}|
  \le
  \omega_{a^*,K}(R_h)\,\|h\|_{L^1},
\]
where
\[
  \omega_{a^*,K}(\rho)
  :=
  \sup_{\substack{x,y\in[-K,K]\\ |x-y|\le\rho}}
  |a^*(x)-a^*(y)|.
\]
Consequently,
\[
  |\alpha_m|
  \le
  \bigl(|a^*(m\Delta)|+\omega_{a^*,K}(R_h)\bigr)\|h\|_{L^1}.
\]
\end{lemma}

\begin{proof}
On $[-R_h,R_h]$ one has $|m\Delta+u-m\Delta|\le R_h$, so
\[
  |a^*(m\Delta+u)-a^*(m\Delta)|\le \omega_{a^*,K}(R_h).
\]
Integrating against $|h(u)|$ gives the first bound, and the second follows by
the triangle inequality.
\end{proof}

\begin{lemma}[A3: packet core/off-core lower bound for the diagonal Archimedean term]\label{lem:packet-alpha-diagonal-core-tail}
For every $r\in(0,R_h]$,
\[
  \alpha_0
  \ge
  m_r^*\,H_r - M_{r,K}^*\,T_r.
\]
\end{lemma}

\begin{proof}
Split the integral defining $\alpha_0$ into the core region $|u|\le r$ and the
tail region $r<|u|\le R_h$:
\[
  \alpha_0
  =
  \int_{|u|\le r} a^*(u)h(u)\,du
  +
  \int_{r<|u|\le R_h} a^*(u)h(u)\,du.
\]
On the core, $a^*(u)\ge m_r^*$; on the tail,
$a^*(u)\ge -M_{r,K}^*$. Since $h\ge0$ for nonnegative $g$, the displayed lower
bound follows.
\end{proof}

\begin{lemma}[A4: crude positive-core diagonal bound]\label{lem:packet-alpha-diagonal-core}
If $R_h\le r$, then
\[
  \alpha_0\ge m_r^*\,\|h\|_{L^1}.
\]
\end{lemma}

\begin{proof}
If $R_h\le r$, then the whole support of $h$ lies inside $[-r,r]$, so
$a^*(u)\ge m_r^*$ on the support of $h$. Integrating gives the bound.
\end{proof}

\begin{lemma}[P1: local prime-mass bound]\label{lem:packet-beta-local-prime-mass}
For every $m\in D(J)$ one has
\[
  0\le \beta_m \le \|h\|_{L^\infty}\,\Pi_K^{\mathrm{sym}}(|m|\Delta;R_h).
\]
\end{lemma}

\begin{proof}
Each summand in the definition of $\beta_m$ is nonnegative because $h\ge0$.
Moreover, $h(\xi_n\mp m\Delta)$ can be nonzero only when
$|\xi_n\mp m\Delta|\le R_h$, and in that case it is bounded by
$\|h\|_{L^\infty}$. Summing over the active nodes yields the claim.
\end{proof}

\begin{lemma}[P2: spacing simplification for the prime mass]\label{lem:packet-beta-spacing}
If $R_h<\delta_K/2$, then every interval $[s-R_h,s+R_h]$ contains at most one
active node from $\Xi_K$. In particular,
\[
  \Pi_K^{\mathrm{sym}}(s;R_h)\le w_{\max}(K),
  \qquad
  \beta_m\le \|h\|_{L^\infty}\,w_{\max}(K).
\]
\end{lemma}

\begin{proof}
The spacing hypothesis implies that two distinct active nodes cannot both lie
inside the same interval of radius $R_h$. The stated prime-mass and
$\beta_m$-bounds then follow from Definition~\ref{def:packet-bound-geometry}
and Lemma~\ref{lem:packet-beta-local-prime-mass}.
\end{proof}

\begin{lemma}[P3: off-diagonal prime vanishing criterion]\label{lem:packet-beta-vanishing}
If $|m|\Delta>R_h$ and
\[
  \operatorname{dist}(|m|\Delta,\Xi_K)>R_h,
\]
then $\beta_m=0$.
\end{lemma}

\begin{proof}
Under the stated distance condition, both
$|\xi_n-|m|\Delta|>R_h$ and $|\xi_n+|m|\Delta|>R_h$ for every active node
$\xi_n\in\Xi_K$. Hence all summands in the definition of $\beta_m$ vanish.
\end{proof}

\begin{proposition}[Finite-symbol envelope]\label{prop:finite-symbol-envelope}
Let $D_+(J):=D(J)\cap\mathbb N$. Since $h$ and $a^*$ are even, one has
$\alpha_{-m}=\alpha_m$ and $\beta_{-m}=\beta_m$, hence for every
$\theta\in[-\pi,\pi]$,
\[
  S_J(\theta)
  \ge
  \alpha_0-\beta_0
  -
  2\sum_{m\in D_+(J)}\bigl(|\alpha_m|+\beta_m\bigr).
\]
In particular, it is sufficient to prove
\[
  \alpha_0-\beta_0
  >
  2\sum_{m\in D_+(J)}\bigl(|\alpha_m|+\beta_m\bigr).
  \tag{C0}
\]
\end{proposition}

\begin{proof}
Write
\[
  S_J(\theta)
  =
  \kappa_0+\sum_{m\in D_+(J)}\kappa_m e^{-im\theta}
  +\sum_{m\in D_+(J)}\kappa_{-m} e^{im\theta}.
\]
Using $\kappa_m=\alpha_m-\beta_m$ and the evenness relations gives
\[
  S_J(\theta)
  =
  \alpha_0-\beta_0
  +
  2\sum_{m\in D_+(J)}(\alpha_m-\beta_m)\cos(m\theta).
\]
Bounding $\cos(m\theta)$ by $1$ in absolute value yields the claimed envelope.
\end{proof}

\begin{proposition}[C1: explicit sufficient inequality for finite admissible dictionaries]\label{prop:packet-explicit-sufficient-criterion}
Fix a finite admissible packet dictionary generated by $g$, $\Delta$, and
$J\subset\mathbb Z$. If there exists $r\in(0,R_h]$ such that
\[
  m_r^*H_r - M_{r,K}^*T_r - \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(0;R_h)
  >
  2\sum_{m\in D_+(J)}
  \Bigl(
    \|a^*\|_{L^\infty(I_m)}\|h\|_{L^1}
    +
    \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(m\Delta;R_h)
  \Bigr),
  \tag{C1}
\]
then
\[
  S_J(\theta)\ge0
  \qquad\text{for all }\theta\in[-\pi,\pi].
\]
Consequently,
\[
  \mathcal Q(\Psi_c*\widetilde{\Psi_c})\ge0
  \qquad\text{for all }c\in\mathbb C^J.
\]
The sharpened approximate-identity version
\[
  m_r^*H_r - M_{r,K}^*T_r - \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(0;R_h)
  >
  2\sum_{m\in D_+(J)}
  \Bigl(
    (|a^*(m\Delta)|+\omega_{a^*,K}(R_h))\|h\|_{L^1}
    +
    \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(m\Delta;R_h)
  \Bigr)
  \tag{C1$'$}
\]
is also sufficient.
\end{proposition}

\begin{proof}
By Lemma~\ref{lem:packet-alpha-diagonal-core-tail},
\[
  \alpha_0\ge m_r^*H_r - M_{r,K}^*T_r.
\]
By Lemma~\ref{lem:packet-beta-local-prime-mass},
\[
  \beta_0\le \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(0;R_h).
\]
For $m\in D_+(J)$, Lemma~\ref{lem:packet-alpha-support-bound} and
Lemma~\ref{lem:packet-beta-local-prime-mass} give
\[
  |\alpha_m|+\beta_m
  \le
  \|a^*\|_{L^\infty(I_m)}\|h\|_{L^1}
  +
  \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(m\Delta;R_h).
\]
Thus \textup{(C1)} implies the sufficient condition \textup{(C0)} from
Proposition~\ref{prop:finite-symbol-envelope}, and hence $S_J\ge0$.
Corollary~\ref{cor:finite-symbol-domination} then gives positivity of
$\mathcal Q(\Psi_c*\widetilde{\Psi_c})$ for all $c$.

The sharpened form \textup{(C1$'$)} is the same argument with the off-diagonal
Archimedean term estimated by Lemma~\ref{lem:packet-alpha-modulus-bound}.
\end{proof}

\begin{corollary}[Sparse-dictionary regime]\label{cor:packet-sparse-dictionary}
Assume
\[
  R_h\le r,
  \qquad
  R_h<\frac{\delta_K}{2},
  \qquad
  \operatorname{dist}(m\Delta,\Xi_K)>R_h
  \quad\text{for all }m\in D_+(J).
\]
Then $\beta_m=0$ for every $m\in D_+(J)$ and the sufficient criterion reduces
to
\[
  m_r^*\,\|h\|_{L^1}
  -
  \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(0;R_h)
  >
  2\sum_{m\in D_+(J)}
  \|a^*\|_{L^\infty(I_m)}\|h\|_{L^1},
  \tag{C2}
\]
or, in the approximate-identity form,
\[
  m_r^*\,\|h\|_{L^1}
  -
  \|h\|_{L^\infty}\Pi_K^{\mathrm{sym}}(0;R_h)
  >
  2\sum_{m\in D_+(J)}
  \bigl(|a^*(m\Delta)|+\omega_{a^*,K}(R_h)\bigr)\|h\|_{L^1}.
  \tag{C2$'$}
\]
\end{corollary}

\begin{proof}
The hypotheses imply the diagonal Archimedean bound of
Lemma~\ref{lem:packet-alpha-diagonal-core} and the off-diagonal prime vanishing
of Lemma~\ref{lem:packet-beta-vanishing}. Substituting these simplifications
into Proposition~\ref{prop:packet-explicit-sufficient-criterion} yields the
claim.
\end{proof}

\begin{proposition}[Canonical centered half-atom bounds]\label{prop:canonical-half-atom-bounds}
Fix $\delta>0$ and $t_0>0$, and set
\[
  g_{\delta,t_0,0}(\xi)
  :=
  \Lambda_\delta(\xi)\rho_{t_0}(\xi)
  =
  \Bigl(1-\frac{|\xi|}{\delta}\Bigr)_+ e^{-4\pi^2 t_0 \xi^2}.
\]
Let
\[
  g:=g_{\delta,t_0,0},
  \qquad
  \widetilde g(x)=g(-x)=g(x),
  \qquad
  h:=g*\widetilde g=g*g.
\]
Then
\[
  R_g=\delta,
  \qquad
  R_h=2\delta,
  \qquad
  \|h\|_{L^1}=\|g\|_{L^1}^2,
  \qquad
  \|h\|_{L^\infty}=h(0)=\|g\|_{L^2}^2.
\]
Writing
\[
  a:=4\pi^2 t_0,
  \qquad
  b:=8\pi^2 t_0,
\]
one has the exact formulas
\[
  \|g\|_{L^1}
  =
  \frac{\sqrt{\pi}}{\sqrt a}\operatorname{erf}(\sqrt a\,\delta)
  -
  \frac{1-e^{-a\delta^2}}{a\delta}
  =
  \frac{1}{2\sqrt{\pi t_0}}\operatorname{erf}(2\pi\sqrt{t_0}\,\delta)
  -
  \frac{1-e^{-4\pi^2 t_0\delta^2}}{4\pi^2 t_0\delta},
\]
and
\[
  \|h\|_{L^\infty}
  =
  \frac{\sqrt{\pi}}{\sqrt b}\operatorname{erf}(\sqrt b\,\delta)
  +
  \frac{\sqrt{\pi}}{2b^{3/2}\delta^2}\operatorname{erf}(\sqrt b\,\delta)
  -
  \frac{2-e^{-b\delta^2}}{b\delta}.
\]
Moreover, for
\[
  M_g(s):=\int_{|x|\le s} g(x)\,dx
  \qquad (0\le s\le \delta)
\]
one has
\[
  M_g(s)
  =
  \frac{\sqrt\pi}{\sqrt a}\operatorname{erf}(\sqrt a\,s)
  -
  \frac{1-e^{-a s^2}}{a\delta},
\]
and therefore
\[
  H_r:=\int_{|u|\le r} h(u)\,du
  =
  \iint_{|x-y|\le r} g(x)g(y)\,dx\,dy
  \ge
  M_g(r/2)^2.
\]
For small $\delta$ one has the asymptotic expansions
\[
  \|g\|_{L^1}
  =
  \delta-\frac{2}{3}\pi^2 t_0\,\delta^3+O(\delta^5),
  \qquad
  \|h\|_{L^1}
  =
  \delta^2+O(\delta^4),
\]
\[
  \|h\|_{L^\infty}
  =
  \frac{2}{3}\delta-\frac{8}{15}\pi^2 t_0\,\delta^3+O(\delta^5),
  \qquad
  H_\delta\ge \frac{9}{16}\delta^2+O(\delta^4).
\]
\end{proposition}

\begin{proof}
The support statements are immediate from the triangular cutoff
$\Lambda_\delta$. Since $g$ is even and nonnegative, so is $h=g*g$, and the
convolution identities
\[
  \|h\|_{L^1}=\|g\|_{L^1}^2,
  \qquad
  h(0)=\int_{\mathbb R} g(x)^2\,dx=\|g\|_{L^2}^2
\]
follow directly. The displayed formulas for $\|g\|_{L^1}$,
$\|h\|_{L^\infty}$, and $M_g(s)$ come from elementary Gaussian integration on
$[0,\delta]$ and $[0,s]$. For the lower bound on $H_r$, restrict the double
integral to the square $|x|,|y|\le r/2$, where automatically $|x-y|\le r$. The
small-$\delta$ expansions follow from the Taylor expansions of
$\operatorname{erf}$ and the Gaussian exponential factors.
\end{proof}

\begin{proposition}[Pilot compact for the canonical half-atom]\label{prop:canonical-half-atom-pilot}
Let
\[
  K:=0.2,
  \qquad
  J:=\{0,1\},
  \qquad
  \Delta:=0.15.
\]
Then the active positive nodes in $[0,K]$ are exactly
\[
  \xi_2=\frac{\log 2}{2\pi}\approx 0.1103,
  \qquad
  \xi_3=\frac{\log 3}{2\pi}\approx 0.17485.
\]
Hence
\[
  \operatorname{dist}(0.15,\Xi_K)\approx 0.02485.
\]
For the canonical packet $g_{\delta,t_0,0}$, if
\[
  \delta<0.0124,
\]
then
\[
  \beta_0=\beta_1=0.
\]
Consequently, on this dictionary
\[
  S_J(\theta)\ge \alpha_0-2|\alpha_1|.
\]
Moreover,
\[
  a^*(0)\approx 33.75,
  \qquad
  a^*(0.15)\approx 13.31,
  \qquad
  a^*(0)-2a^*(0.15)\approx 7.13>0.
\]
Therefore, for $\delta$ small enough that the modulus error term from
Lemma~\ref{lem:packet-alpha-modulus-bound} stays below this numerical gap, one
has
\[
  \alpha_0>2|\alpha_1|,
\]
hence
\[
  S_J(\theta)\ge0
  \qquad\text{for all }\theta\in[-\pi,\pi],
\]
and therefore
\[
  \mathcal Q(\Psi_c*\widetilde{\Psi_c})\ge0
  \qquad\text{for all }c\in\mathbb C^2.
\]
\end{proposition}

\begin{proof}
Since $e^{2\pi K}<4$, the only active positive nodes are $\xi_2$ and $\xi_3$.
If $2\delta<\operatorname{dist}(0.15,\Xi_K)$, then the off-diagonal prime
coefficient vanishes by Lemma~\ref{lem:packet-beta-vanishing}, so $\beta_1=0$.
If also $2\delta<\xi_2$, then the diagonal prime coefficient vanishes as well,
so $\beta_0=0$. Both conditions hold for $\delta<0.0124$.

The finite symbol for $J=\{0,1\}$ is
\[
  S_J(\theta)=\kappa_0+\kappa_1 e^{-i\theta}+\kappa_{-1}e^{i\theta}
  =(\alpha_0-\beta_0)+2(\alpha_1-\beta_1)\cos\theta,
\]
so the previous vanishing gives
\[
  S_J(\theta)\ge \alpha_0-2|\alpha_1|.
\]
Finally, Lemma~\ref{lem:packet-alpha-modulus-bound} gives
\[
  \alpha_0=a^*(0)\|h\|_{L^1}+O(\omega_{a^*,K}(2\delta)\|h\|_{L^1}),
  \qquad
  \alpha_1=a^*(0.15)\|h\|_{L^1}+O(\omega_{a^*,K}(2\delta)\|h\|_{L^1}),
\]
and the numerical gap $a^*(0)-2a^*(0.15)>0$ therefore implies
$\alpha_0>2|\alpha_1|$ once $\delta$ is sufficiently small.
\end{proof}

\begin{theorem}[P8 / PSD-pd: conditional corrected compact positivity via finite-symbol verification]\label{thm:PSD-pd}
Fix a compact window $[-K,K]$ and let
$\mathcal P_{K,\mathrm{tr}}(t_0)\subseteq \mathcal P_K(t_0)$ be a
translation-compatible packet subspace such that:
\begin{enumerate}
  \item every $\Psi\in\mathcal P_{K,\mathrm{tr}}(t_0)$ belongs to some finite
  admissible translation dictionary generated by one base packet $g$ and one
  step $\Delta$;
  \item
  \[
    \overline{
      \operatorname{cone}\{\Psi*\widetilde{\Psi}:\ \Psi\in\mathcal P_{K,\mathrm{tr}}(t_0)\}
    }^{\|\cdot\|_\infty}
    =
    \mathcal W_K^{\mathrm{pd}}.
  \]
\end{enumerate}
Assume moreover that for every finite admissible translation dictionary
appearing in $\mathcal P_{K,\mathrm{tr}}(t_0)$ the hypotheses of
Corollary~\ref{cor:regularized-p7} hold. Equivalently, it is enough to verify
the explicit coefficient inequalities of Proposition~\ref{prop:packet-explicit-sufficient-criterion}
or its sparse-dictionary Corollary~\ref{cor:packet-sparse-dictionary} on every
such admissible dictionary.
Then
\[
  Q^\star(t;\Phi)\ge0\qquad\text{for every }\Phi\in \mathcal W_K^{\mathrm{pd}}.
\]
\end{theorem}

\begin{proof}
By Corollary~\ref{cor:regularized-p7},
\[
  \mathcal Q(\Psi*\widetilde{\Psi})\ge0
  \qquad\text{for every }\Psi\in\mathcal P_{K,\mathrm{tr}}(t_0).
\]
Since $\Psi*\widetilde{\Psi}$ is even, this reads
\[
  Q^\star(t;\Psi*\widetilde{\Psi})\ge0
  \qquad\text{for every }\Psi\in\mathcal P_{K,\mathrm{tr}}(t_0).
\]
Hence positivity holds on the dense packet-autocorrelation cone
$\operatorname{cone}\{\Psi*\widetilde{\Psi}:\ \Psi\in\mathcal P_{K,\mathrm{tr}}(t_0)\}$.
By the density assumption on $\mathcal P_{K,\mathrm{tr}}(t_0)$ and by Corollary~\ref{cor:A2-Lip},
$Q^\star(t;\cdot)$ is continuous on the ambient compact-support class $B_K$,
hence on $\mathcal W_K^{\mathrm{pd}}\subseteq B_K$. Therefore positivity extends
to all of $\mathcal W_K^{\mathrm{pd}}$.
\end{proof}

\begin{definition}[Compact spectral weight]\label{def:compact-spectral-weight}
Fix $K>0$ and define
\[
  \mathcal Q_K(F)
  :=
  \int_{\mathbb R} a_K^*(\xi)F(\xi)\,d\xi
  -
  \frac12\sum_{\xi_n\in\Xi_K}\frac{2\Lambda(n)}{\sqrt n}\bigl(F(\xi_n)+F(-\xi_n)\bigr),
\]
\[
  a_K^*(\xi):=a^*(\xi)\mathbf 1_{[-K,K]}(\xi),
  \qquad
  \Xi_K:=\{\xi_n:\ 0<\xi_n\le K\}.
\]
If $\operatorname{supp}F\subset[-K,K]$, then $\mathcal Q_K(F)=\mathcal Q(F)$, and
if $F$ is even, then $\mathcal Q_K(F)=Q^\star(t;F)$.
With the Fourier convention
\[
  \widehat f(u):=\int_{\mathbb R} f(x)e^{-iux}\,dx,
  \qquad
  f(x)=\frac1{2\pi}\int_{\mathbb R}\widehat f(u)e^{iux}\,du,
\]
define the compact spectral weight
\[
  W_K(u)
  :=
  \widehat{a_K^*}(u)
  -
  \sum_{\xi_n\in\Xi_K}\frac{2\Lambda(n)}{\sqrt n}\cos(u\xi_n).
\]
\end{definition}

\begin{proposition}[S1: exact compact spectral identity]\label{prop:compact-spectral-identity}
Let $\Psi\in C_c([-K/2,K/2])$ and set
\[
  \widetilde\Psi(x):=\overline{\Psi(-x)},
  \qquad
  F_\Psi:=\Psi*\widetilde{\Psi}.
\]
Then
\[
  F_\Psi\in C_c([-K,K]),
\]
\[
  \widehat{F_\Psi}(u)=|\widehat\Psi(u)|^2
\]
and
\[
  \mathcal Q_K(F_\Psi)
  =
  \frac1{2\pi}\int_{\mathbb R} W_K(u)\,|\widehat\Psi(u)|^2\,du.
\]
\end{proposition}

\begin{proof}
Since $\Psi$ is compactly supported in $[-K/2,K/2]$, the convolution
$F_\Psi=\Psi*\widetilde{\Psi}$ is supported in $[-K,K]$. Therefore
\[
  \mathcal Q_K(F_\Psi)
  =
  \int_{\mathbb R} a_K^*(\xi)F_\Psi(\xi)\,d\xi
  -
  \frac12\sum_{\xi_n\in\Xi_K}\frac{2\Lambda(n)}{\sqrt n}
  \bigl(F_\Psi(\xi_n)+F_\Psi(-\xi_n)\bigr),
\]
because $F_\Psi$ is even. The Fourier identity
\[
  \widehat{F_\Psi}(u)=\widehat\Psi(u)\,\overline{\widehat\Psi(u)}
  =|\widehat\Psi(u)|^2
\]
is standard for autocorrelations. Applying Fourier inversion to the compactly
supported Archimedean term gives
\[
  \int_{\mathbb R} a_K^*(\xi)F_\Psi(\xi)\,d\xi
  =
  \frac1{2\pi}\int_{\mathbb R}\widehat{a_K^*}(u)\,|\widehat\Psi(u)|^2\,du.
\]
For each active node $\xi_n\in\Xi_K$,
\[
  F_\Psi(\pm\xi_n)
  =
  \frac1{2\pi}\int_{\mathbb R} e^{\pm iu\xi_n}|\widehat\Psi(u)|^2\,du.
\]
Hence
\[
  \frac12\bigl(F_\Psi(\xi_n)+F_\Psi(-\xi_n)\bigr)
  =
  \frac1{2\pi}\int_{\mathbb R} \cos(u\xi_n)\,|\widehat\Psi(u)|^2\,du.
\]
Summing over $\Xi_K$ gives the claimed identity.
\end{proof}

\begin{proposition}[S2: spectral positivity criterion on a compact]\label{prop:compact-spectral-positivity}
If
\[
  W_K(u)\ge0
  \qquad\text{for every }u\in\mathbb R,
\]
then
\[
  \mathcal Q_K(\Psi*\widetilde{\Psi})\ge0
  \qquad\text{for every }\Psi\in C_c([-K/2,K/2]).
\]
Equivalently, the packet kernel
\[
  K_{Q,K}(\Psi,\Phi):=\mathcal Q_K(\Psi*\widetilde{\Phi})
\]
is positive semidefinite on $C_c([-K/2,K/2])$, i.e.
\[
  \sum_{i,j=1}^N c_i\overline{c_j}\,
  \mathcal Q_K(\Psi_i*\widetilde{\Psi_j})
  \ge0
\]
for every finite family $\Psi_1,\dots,\Psi_N\in C_c([-K/2,K/2])$ and every
$c_1,\dots,c_N\in\mathbb C$.
\end{proposition}

\begin{proof}
The scalar inequality follows immediately from
Proposition~\ref{prop:compact-spectral-identity}. Positive semidefiniteness of
the sesquilinear kernel follows by letting
\[
  \Psi_c:=\sum_{i=1}^N c_i\Psi_i
\]
for a finite family $\Psi_1,\dots,\Psi_N$ and using
\[
  \sum_{i,j=1}^N c_i\overline{c_j}\,
  \mathcal Q_K(\Psi_i*\widetilde{\Psi_j})
  =
  \mathcal Q_K(\Psi_c*\widetilde{\Psi_c})\ge0.
\]
\end{proof}

\begin{theorem}[S3: conditional corrected compact positivity]\label{thm:compact-spectral-closure}
Define the corrected local positive-definite cone
\[
  \mathcal W_K^{\mathrm{pd}}
  :=
  \overline{
    \operatorname{cone}\{\Psi*\widetilde\Psi:\ \Psi\in C_c([-K/2,K/2])\}
  }^{\|\cdot\|_\infty}.
\]
Let $\mathcal P_K\subset C_c([-K/2,K/2])$ be a pre-packet space such that
\[
  \overline{
    \operatorname{cone}\{\Psi*\widetilde\Psi:\ \Psi\in\mathcal P_K\}
  }^{\|\cdot\|_\infty}
  =
  \mathcal W_K^{\mathrm{pd}}.
\]
Assume, moreover, that $\mathcal Q_K$ is continuous on

\begin{theorem}[H1$^{\mathrm f}$: exact filtered bulk intertwining]\label{thm:H1-filtered-bulk}
Fix $a>0$. Assume there exist an integer $N_a\ge 0$ and a scalar
$\kappa(a)>0$ such that the two direct filtered bulk identities
\begin{equation}\label{eq:H1-filtered-bulk-plus-plus}
  M_{mn}^{++}(a)=\kappa(a)\,\widetilde q_{mn}^{++},
  \qquad m,n>N_a,
\end{equation}
and
\begin{equation}\label{eq:H1-filtered-bulk-plus-minus}
  M_{mn}^{+-}(a)=\kappa(a)\,\widetilde q_{mn}^{+-},
  \qquad m,n>N_a,
\end{equation}
hold, where the filtered Suzuki entries $M_{mn}^{\sigma\tau}(a)$ and the
filtered Section~8 entries $\widetilde q_{mn}^{\sigma\tau}$ are defined by
Proposition~\ref{prop:H1-raw-entry-reduction} and
Proposition~\ref{prop:H1-filtered-q-blocks}.

Then for every $M>N_a$ one has the exact filtered operator identity
\[
  S_{a,M,N_a}^*\,G_g[a]\,S_{a,M,N_a}
  =
  \kappa(a)\,\widetilde Q_{M,N_a}.
\]
Equivalently, for every $N_a<n,m\le M$ and every $\sigma,\tau\in\{+,-\}$,
\[
  \bigl\langle G_g[a]\phi_n^\sigma[a],\phi_m^\tau[a]\bigr\rangle_{L^2(-a,a)}
  =
  \kappa(a)\,\bigl(\widetilde Q_{M,N_a}^{\sigma\tau}\bigr)_{mn}.
\]
\end{theorem}

\begin{proof}[Proof skeleton]
The hypotheses \eqref{eq:H1-filtered-bulk-plus-plus}
and~\eqref{eq:H1-filtered-bulk-plus-minus} cover the filtered families
$(+,+)$ and $(+,-)$. By Corollary~\ref{cor:H1-bulk-symmetry-reduction}, the
remaining filtered families $(-,-)$ and $(-,+)$ follow from conjugation and
self-adjoint symmetry:
\[
  M_{mn}^{--}(a)=\overline{M_{nm}^{++}(a)},
  \qquad
  M_{mn}^{-+}(a)=\overline{M_{nm}^{+-}(a)},
\]
and the same Hermitian relations hold on the Q3 side for the filtered entries
$\widetilde q_{mn}^{\sigma\tau}$.
Hence for every $\sigma,\tau\in\{+,-\}$ and every $N_a<n,m\le M$,
\[
  M_{mn}^{\sigma\tau}(a)=\kappa(a)\,\widetilde q_{mn}^{\sigma\tau}.
\]
Thus the matrices of $S_{a,M,N_a}^*G_g[a]S_{a,M,N_a}$ and
$\kappa(a)\widetilde Q_{M,N_a}$ agree entrywise in the two-sided tail basis of
$\mathcal P_{M,N_a}$, so the corresponding operators are equal.

The exact metric pullback
\[
  S_{a,M,N_a}^*J_aS_{a,M,N_a}=B_{M,N_a}=\Delta_{M,N_a}^*\Delta_{M,N_a}
\]
is already provided by Proposition~\ref{prop:Ja-pullback-filtered-tail}; it is
not part of the present bulk-intertwining step, but is the metric input used in
Theorem~\ref{thm:H3-filtered-transfer}. Therefore the whole burden of
Theorem~\ref{thm:H1-filtered-bulk} is exactly the pair of direct filtered bulk
identities \eqref{eq:H1-filtered-bulk-plus-plus}
and~\eqref{eq:H1-filtered-bulk-plus-minus}.
\end{proof}

  -
  \sum_{\substack{n\ge2\\ |\xi_n|\le B}}
  w(n)\,\Phi_{B,t}(\xi_n)\,e^{-2\pi i(r-s)\xi_n}
\]
be the stable raw-compressed Section~8 entries from
Proposition~\ref{prop:H1-raw-q-formula}. Then there is no scalar $\kappa(a)$
and no integer $N_a$ such that
\[
  w_{mn}(a)=\kappa(a)\,q_{mn}
  \qquad\text{for all }m,n>N_a.
\]
Consequently, the raw bulk target
\[
  w_{mn}(a)=\kappa(a)q_{mn},
  \qquad
  w_{m,-n}(a)=\kappa(a)q_{m,-n},
\]
cannot serve as the exact bulk hypothesis for Theorem~\ref{thm:H1-filtered-bulk}.
\end{proposition}

\begin{proof}
The raw Q3 matrix is Toeplitz:
\[
  q_{rs}=q(r-s),
\]
hence its diagonal is constant:
\[
  q_{rr}=a_0-\sum_{\substack{n\ge2\\ |\xi_n|\le B}}
  w(n)\,\Phi_{B,t}(\xi_n).
\]
On the Suzuki--Yoshida side, the raw Weil matrix in the basis
$\{\chi_n[a]\}_{n\in\mathbb Z}$ has logarithmically growing diagonal and
decaying off-diagonal tails; in particular,
\[
  |w_{nn}(a)|\sim \log|n|
  \qquad(|n|\to\infty).
\]
Therefore no fixed scalar multiple of the constant diagonal of $q_{rs}$ can
match the unbounded diagonal of $w_{rs}(a)$ on the bulk.
\end{proof}

\begin{lemma}[Fourier transforms of the adjacent Suzuki tails]\label{lem:H1-adjacent-fourier}
For $n\ge 1$,
\[
  \widehat{\chi_n[a]}(z)
  =
  (-1)^n\sqrt{\frac2a}\,\frac{\sin(az)}{z+\alpha_n},
\]
and hence
\[
  \widehat{\psi_n^\sigma[a]}(z)
  =
  (-1)^n\sqrt{\frac2a}\,\sin(az)
  \left(
    \frac{1}{z+\varepsilon_\sigma\alpha_n}
    -
    \frac{1}{z+\varepsilon_\sigma\alpha_{n+1}}
  \right),
  \qquad
  \sigma\in\{+,-\}.
\]
Equivalently,
\[
  \widehat{\psi_n^\sigma[a]}(z)
  =
  (-1)^n\sqrt{\frac2a}\,\frac{\pi}{a}\,
  \frac{\sin(az)}
  {(z+\varepsilon_\sigma\alpha_n)(z+\varepsilon_\sigma\alpha_{n+1})}.
\]
\end{lemma}

\begin{proposition}[Raw-entry reduction of the filtered bulk match]\label{prop:H1-raw-entry-reduction}
Define the antiderivative packets
\[
  \psi_n^+[a]:=I_0^{(a)}\phi_n^+[a]=\chi_n[a]+\chi_{n+1}[a],
  \qquad
  \psi_n^-[a]:=I_0^{(a)}\phi_n^-[a]=\chi_{-n}[a]+\chi_{-(n+1)}[a].
\]
For $\sigma,\tau\in\{+,-\}$ and $N<n,m\le M$, set
\[
  M_{mn}^{\sigma\tau}(a)
  :=
  \bigl\langle G_g[a]\phi_n^\sigma[a],\phi_m^\tau[a]\bigr\rangle_{L^2(-a,a)}
  =
  W\bigl(\psi_n^\sigma[a]*\widetilde{\psi_m^\tau[a]}\bigr).
\]
Then
\[
  M_{mn}^{\sigma\tau}(a)
  =
  w_{\varepsilon_\sigma m,\varepsilon_\tau n}(a)
  +
  w_{\varepsilon_\sigma(m+1),\varepsilon_\tau n}(a)
  +
  w_{\varepsilon_\sigma m,\varepsilon_\tau(n+1)}(a)
  +
  w_{\varepsilon_\sigma(m+1),\varepsilon_\tau(n+1)}(a).
\]
Equivalently, each filtered bulk block is the $(1+z)$ / $(1+z^{-1})$
four-term stencil of the corresponding raw Weil entries.
\end{proposition}

\begin{corollary}[Raw $\gamma$-sum form of the Suzuki blocks]\label{cor:H1-raw-gamma-blocks}
For $m,n\ge 1$ and $\sigma,\tau\in\{+,-\}$,
\[
  M_{mn}^{\sigma\tau}(a)
  =
  \frac{2\pi^2}{a^3}(-1)^{m+n}
  \sum_{\gamma}
  \frac{\sin^2(a\gamma)}
  {(\gamma-\varepsilon_\sigma\alpha_m)
   (\gamma-\varepsilon_\sigma\alpha_{m+1})
   (\gamma+\varepsilon_\tau\alpha_n)
   (\gamma+\varepsilon_\tau\alpha_{n+1})},
\]
where $\gamma$ runs over the zeros of $\xi(1/2-iz)$ with multiplicity.
In particular,
\[
  M_{mn}^{++}(a)
  =
  \frac{2\pi^2}{a^3}(-1)^{m+n}
  \sum_\gamma
  \frac{\sin^2(a\gamma)}
  {(\gamma-\alpha_m)(\gamma-\alpha_{m+1})(\gamma+\alpha_n)(\gamma+\alpha_{n+1})},
\]
\[
  M_{mn}^{+-}(a)
  =
  \frac{2\pi^2}{a^3}(-1)^{m+n}
  \sum_\gamma
  \frac{\sin^2(a\gamma)}
  {(\gamma-\alpha_m)(\gamma-\alpha_{m+1})(\gamma-\alpha_n)(\gamma-\alpha_{n+1})}.
\]
\end{corollary}

\begin{proposition}[Exact filtered Section~8 blocks]\label{prop:H1-filtered-q-blocks}
For $m,n\ge 1$ and $\sigma,\tau\in\{+,-\}$, define
\[
  \widetilde q_{mn}^{\sigma\tau}
  :=
  q_{\varepsilon_\sigma m,\varepsilon_\tau n}
  +
  q_{\varepsilon_\sigma(m+1),\varepsilon_\tau n}
  +
  q_{\varepsilon_\sigma m,\varepsilon_\tau(n+1)}
  +
  q_{\varepsilon_\sigma(m+1),\varepsilon_\tau(n+1)}.
\]
Then the filtered finite section is exactly
\[
  \widetilde Q_{M,N}=\Delta_{M,N}^*Q_{M+1}\Delta_{M,N}
\]
in the two-sided tail basis, with block entries
$\widetilde q_{mn}^{\sigma\tau}$.
If one writes
\[
  q_{rs}=a_{r-s}-p_{rs},
\]
with
\[
  p_{rs}
  :=
  \sum_{\substack{n\ge2\\ |\xi_n|\le B}}
  w(n)\,\Phi_{B,t}(\xi_n)\,e^{-2\pi i(r-s)\xi_n},
\]
then
\[
  \widetilde q_{mn}^{\sigma\tau}
  =
  \widetilde a_{mn}^{\sigma\tau}
  -
  \widetilde p_{mn}^{\sigma\tau},
\]
where
\[
  \widetilde a_{mn}^{\sigma\tau}
  =
  a_{d_{\sigma\tau}(m,n)}
  +
  a_{d_{\sigma\tau}(m,n)+\varepsilon_\sigma}
  +
  a_{d_{\sigma\tau}(m,n)-\varepsilon_\tau}
  +
  a_{d_{\sigma\tau}(m,n)+\varepsilon_\sigma-\varepsilon_\tau},
\]
\[
  \widetilde p_{mn}^{\sigma\tau}
  =
  p_{\varepsilon_\sigma m,\varepsilon_\tau n}
  +
  p_{\varepsilon_\sigma(m+1),\varepsilon_\tau n}
  +
  p_{\varepsilon_\sigma m,\varepsilon_\tau(n+1)}
  +
  p_{\varepsilon_\sigma(m+1),\varepsilon_\tau(n+1)},
\]
and
\[
  d_{\sigma\tau}(m,n):=\varepsilon_\sigma m-\varepsilon_\tau n.
\]
In particular,
\[
  \widetilde a_{mn}^{++}
  =
  2a_{m-n}+a_{m-n+1}+a_{m-n-1},
\]
\[
  \widetilde a_{mn}^{+-}
  =
  a_{m+n}+2a_{m+n+1}+a_{m+n+2},
\]
\[
  \widetilde a_{mn}^{-+}
  =
  a_{-(m+n)}+2a_{-(m+n+1)}+a_{-(m+n+2)},
\]
\[
  \widetilde a_{mn}^{--}
  =
  2a_{n-m}+a_{n-m+1}+a_{n-m-1}.
\]
\end{proposition}

\begin{corollary}[Symmetry reduction of the bulk match]\label{cor:H1-bulk-symmetry-reduction}
One has
\[
  \phi_n^-[a]=\overline{\phi_n^+[a]},
  \qquad
  \psi_n^-[a]=\overline{\psi_n^+[a]}.
\]
Since $G_g[a]$ is self-adjoint with Hermitian kernel,
\[
  M_{mn}^{--}(a)=\overline{M_{nm}^{++}(a)},
  \qquad
  M_{mn}^{-+}(a)=\overline{M_{nm}^{+-}(a)}.
\]
Assuming the same Hermitian normalization on the Q3 side,
\[
  \widetilde q_{mn}^{--}=\overline{\widetilde q_{nm}^{++}},
  \qquad
  \widetilde q_{mn}^{-+}=\overline{\widetilde q_{nm}^{+-}}.
\]
Therefore the direct filtered bulk target may be reduced to the two filtered
families
\[
  M_{mn}^{++}(a)=\kappa(a)\,\widetilde q_{mn}^{++},
  \qquad
  M_{mn}^{+-}(a)=\kappa(a)\,\widetilde q_{mn}^{+-},
  \qquad
  m,n>N_a,
\]
and then the remaining filtered blocks follow formally by Hermitian symmetry.
\end{corollary}

\noindent
The live bulk theorem is therefore no longer the abstract existence of
$S_{a,M,N}$ and $J_a$, nor any raw $\chi_n[a]$-level identity. By
Proposition~\ref{prop:H1-raw-bulk-mismatch}, the raw basis is only a diagnostic
layer and cannot support the exact bridge. The narrowest active target is now
the filtered bulk classifier on the two families $(+,+)$ and $(+,-)$; the
remaining filtered block equalities are formal consequences of Hermitian
symmetry, and after that filtered bulk classification the only remaining bridge
brick is the finite-dimensional Suzuki cap, possibly augmented by the same
structured small-rank correction.

\begin{theorem}[H2$^{\mathrm f}$: Suzuki tail/cap reduction]\label{thm:H2-filtered-cap}
Fix $a>0$ and $N_a$ as in Theorem~\ref{thm:H1-filtered-bulk}. Define the closed tail space
\[
  V_a^{\mathrm{tail}}
  :=
  \overline{\bigcup_{M>N_a} S_{a,M,N_a}\mathcal P_{M,N_a}}^{\,L^2(-a,a)}.
\]
Assume there exists a finite-dimensional subspace
$A_a^{\mathrm{cap}}\subset L^2(-a,a)$ such that
\[
  L^2(-a,a)=V_a^{\mathrm{tail}}\oplus A_a^{\mathrm{cap}},
\]
and the decomposition is $q_{G,a}$-orthogonal:
\[
  \langle G_g[a]v,w\rangle_{L^2(-a,a)}=0
  \qquad
  (v\in V_a^{\mathrm{tail}},\ w\in A_a^{\mathrm{cap}}).
\]
Assume moreover that $A_a^{\mathrm{cap}}$ is spanned by finitely many
annihilator vectors
\[
  u_n[a]-v_n[a],
  \qquad |n|\le N_a+1,
\]
with $v_n[a]\in V_a^{\mathrm{tail}}$,
so that the restriction of $q_{G,a}$ to $A_a^{\mathrm{cap}}$
is represented by a finite Hermitian matrix
\[
  H_a^{\mathrm{cap}}.
\]
\end{theorem}

\begin{theorem}[H3$^{\mathrm f}$: filtered gap transfer]\label{thm:H3-filtered-transfer}
Assume Theorems~\ref{thm:H1-filtered-bulk}
and~\ref{thm:H2-filtered-cap}.
Suppose there exists $c(a)>0$ such that
\[
  Q_M\ge c(a)\,I_{P_M}
  \qquad\text{for every }M\ge N_a+1.
\]
Then for every $M>N_a$,
\[
  \widetilde Q_{M,N_a}\ge c(a)\,B_{M,N_a}.
\]
Hence,
\[
  q_{G,a}(v)\ge \kappa(a)c(a)\,q_{J,a}(v)
  \qquad\text{for every }v\in V_a^{\mathrm{tail}}.
\]
If, in addition, the finite cap matrix $H_a^{\mathrm{cap}}$ is positive definite,
then
\[
  \ker G_g[a]=\{0\}.
\]
Equivalently, $0$ is not an eigenvalue of $G_g[a]$.
\end{theorem}

\begin{corollary}[H4$^{\mathrm f}$: RH from the filtered Suzuki--Q3 bridge]\label{cor:H4-filtered-RH}
Assume that for every $a>0$ the hypotheses of
Theorems~\ref{thm:H1-filtered-bulk},
\ref{thm:H2-filtered-cap},
and~\ref{thm:H3-filtered-transfer}
hold.
Then $0$ is not an eigenvalue of $G_g[a]$ for every $a>0$.
Hence the Riemann Hypothesis is true.
\end{corollary}

\begin{remark}[Semilocal layer as an engineering refinement only]\label{rem:h1-semilocal-engine}
The semilocal Connes--Consani--Moscovici machinery remains useful only as a
finite-prime basis/Gram supplier for \textup{H1}$^{\mathrm f}$. It may provide a
canonical packet family, a Jacobi basis, or auxiliary Gram data feeding the
matrix comparison
\[
  \bigl[\langle G_g[a]\phi_n^\sigma[a],\phi_m^\tau[a]\rangle\bigr]
  \qquad\text{vs.}\qquad
  \kappa(a)\widetilde Q_{M,N}^{\sigma\tau},
\]
but it is not a separate RH endgame. The live bridge remains the two-sided
filtered Suzuki package
\[
  \textup{H1}^{\mathrm f}\Longrightarrow \textup{H2}^{\mathrm f}
  \Longrightarrow \textup{H3}^{\mathrm f}\Longrightarrow \textup{H4}^{\mathrm f},
\]
with the finite-dimensional Suzuki cap handled only after the bulk
intertwining is in place.
\end{remark}

\begin{remark}[Compact scalar obstruction]\label{rem:compact-scalar-obstruction}
The compact spectral package \textup{(S1)}--\textup{(S4)} is mathematically
correct as a compact-truncation reduction, but it is no longer a viable public
mainline once $\Xi_K\neq\varnothing$. Indeed, $a_K^*\in L^1(\mathbb R)$, so
\[
  \widehat{a_K^*}(u)\to0 \qquad (|u|\to\infty)
\]
by the Riemann--Lebesgue lemma. On the other hand, the prime contribution
\[
  \sum_{\xi_n\in\Xi_K}\frac{2\Lambda(n)}{\sqrt n}\cos(u\xi_n)
\]
is a finite positive cosine sum, and by simultaneous approximation it returns
arbitrarily close to its full positive mass infinitely often. Therefore the
pointwise target $W_K(u)\ge0$ cannot serve as the live compact mainline on any
compact with active nodes. In the current draft \textup{(S1)}--\textup{(S4)}
are retained only as a correct diagnostic reduction.
\end{remark}

\begin{remark}[Why shifted A1$'$ is now auxiliary]\label{rem:G1-frozen-target}
The shifted theorem A1$'$ remains mathematically valuable: it proves density on the broad restriction cone $R_K$ and it still supplies approximation technology. But after the target-cone audit it no longer serves as the public RH density theorem, because the corrected target is the positive-definite cone $\mathcal W_K^{\mathrm{pd}}$, not the full broad cone of even nonnegative compact-support tests.
\end{remark}
