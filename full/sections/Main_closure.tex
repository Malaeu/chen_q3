\subsection{Main closure: from analytic modules to Weil positivity}\label{sec:Main-closure}

\subsection*{Standing contract}
For each compact window $[-K,K]$ we distinguish:
\[
  R_K := C^+_{\mathrm{even}}([-K,K]),
  \qquad
  W_K := \{\Phi\in C_c(\mathbb R): \operatorname{supp}\Phi\subset[-K,K],\ \Phi\ \text{even},\ \Phi\ge 0\}.
\]
The current draft already supplies:
\begin{itemize}
  \item \textbf{(T0)} the Guinand--Weil crosswalk, cf.\ Proposition~\ref{prop:T0-GW};
  \item \textbf{(A1$'$)} density of shifted evenized Fej\'er$\times$heat atoms on the restriction cone $R_K$, cf.\ Theorem~\ref{a1:thm:A1-local-density};
  \item \textbf{(A2)} continuity of $Q^\star(t;\cdot)$ on the admissible support cone $W_K$, cf.\ Section~\ref{sec:A2};
  \item \textbf{(A3)+(RKHS)} the centered positivity mechanism at the critical scale.
\end{itemize}

What remains unresolved are the closure gates:
\begin{itemize}
  \item \textbf{(G1) Support upgrade.} Upgrade the density input from $R_K$ to admissible $W_K$.
  \item \textbf{(G2) Common family.} Fix one exact admissible generator family $\mathcal G_K\subseteq W_K$ implementing the `G1` route.
  \item \textbf{(G3) Positivity on $\mathcal G_K$.} Prove $Q^\star(t;\Phi)\ge 0$ on that same family.
  \item \textbf{(G5) LF compatibility.} The compact-by-compact positivity statements must be compatible with the natural embeddings $W_K\hookrightarrow W_{K'}$.
\end{itemize}

\begin{theorem}[Conditional main positivity on $\mathcal W$]\label{thm:Main-positivity}
Assume \textup{(T0)}, \textup{(A2)}, and for every compact window $[-K,K]$ a cone $\mathcal G_K\subseteq W_K$ such that
\[
  \overline{\mathcal G_K}^{\|\cdot\|_\infty}=W_K
  \qquad\text{and}\qquad
  Q^\star(t;\Phi)\ge0\ \text{ for every }\Phi\in\mathcal G_K.
\]
Assume moreover that these local positivity statements are compatible with the natural embeddings $W_K\hookrightarrow W_{K'}$. Then
\[
  Q^\star(t;\Phi)\ \ge\ 0\qquad\text{for every even, real, compactly supported }\Phi\in \mathcal W,
\]
where $\mathcal W=\bigcup_{K>0}W_K$ is the Weil cone from Section~\ref{sec:notation}.
\end{theorem}

\begin{proof}
Fix $K>0$. By Proposition~\ref{prop:compatibility-reduction-shifted-atoms}, the assumptions imply
\[
  Q^\star(t;\Phi)\ge 0 \qquad \text{for every }\Phi\in W_K.
\]
Now let $\Phi\in\mathcal W$. Choose $K$ with $\operatorname{supp}\Phi\subset[-K,K]$. Then $\Phi\in W_K$, hence $Q^\star(t;\Phi)\ge 0$. The compatibility assumption ensures that the conclusion is independent of the chosen compact exhaustion. Therefore $Q^\star(t;\Phi)\ge0$ on the Weil cone $\mathcal W$.
\end{proof}

\begin{remark}[Where the present draft stops]
The current analytic core supplies two separate ingredients:
\begin{enumerate}
  \item the density theorem A1$'$ on the restriction cone $R_K$ for shifted evenized Fej\'er$\times$heat atoms;
  \item the centered positivity mechanism coming from A3 and the single-scale RKHS cap.
\end{enumerate}
What is not yet identified in the paper is a \emph{single admissible} family $\mathcal G_K\subseteq W_K$ that both implements the support upgrade from $R_K$ and is already covered by the positivity theorem. The unresolved work is therefore exactly `G1-G3`, not another LF theorem.
\end{remark}

\begin{proposition}[Compact closure from a common admissible family]\label{prop:compatibility-reduction-shifted-atoms}
Fix a compact window $[-K,K]$ and assume that $\mathcal G_K\subseteq W_K$ satisfies
\[
  \overline{\mathcal G_K}^{\|\cdot\|_\infty}=W_K
  \qquad\text{and}\qquad
  Q^\star(t;\Phi)\ge0\ \text{ for every }\Phi\in\mathcal G_K.
\]
Then
\[
  Q^\star(t;\Phi)\ge 0 \qquad \text{for every }\Phi\in W_K.
\]
\end{proposition}

\begin{proof}
By the assumed atomwise inequality and linearity of $Q^\star$, the functional is nonnegative on the cone generated by $\mathcal G_K$.
By Corollary~\ref{cor:A2-Lip}, $Q^\star(t;\cdot)$ is continuous on $W_K$ in the uniform norm.
Passing to the uniform limit along the dense cone $\mathcal G_K$ gives $Q^\star(t;\Phi)\ge 0$ for every $\Phi\in W_K$.
\end{proof}

\begin{remark}[Exact remaining mathematical task]\label{rem:compatibility-missing-scalar}
Proposition~\ref{prop:compatibility-reduction-shifted-atoms} shows that the closure step on a fixed compact is routine once a common admissible family $\mathcal G_K$ has been identified. The unresolved work is therefore:
\begin{enumerate}
  \item build the support upgrade from the restriction cone $R_K$ to admissible $W_K$;
  \item freeze one exact admissible family $\mathcal G_K\subseteq W_K$ implementing that upgrade;
  \item prove positivity on that same $\mathcal G_K$.
\end{enumerate}
If future work chooses $\mathcal G_K$ to be a support-compatible realization of the shifted evenized atoms from A1$'$, then the remaining scalar task becomes positivity on those admissible generators. Until such a family is fixed, phrases like “density + continuity imply positivity on $W_K$” remain only a conditional template.
\end{remark}
