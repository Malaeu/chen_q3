\subsection{Weil linkage: positivity on the corrected cone implies the Riemann Hypothesis}\label{sec:Weil-linkage}

\begin{theorem}[T0-pd / corrected Weil linkage]\label{thm:Weil-criterion}
Let $Q^\star(t;\cdot)$ be the Weil functional attached to $\zeta(s)$ in the normalization of Section~\ref{sec:T0}. Write
\[
  \widetilde{\psi}(x):=\overline{\psi(-x)},
  \qquad
  \mathcal W^{\mathrm{pd}}:=\varinjlim_{K\to\infty}\mathcal W_K^{\mathrm{pd}},
\]
where each local cone $\mathcal W_K^{\mathrm{pd}}$ is generated by convolution squares
$\psi*\widetilde{\psi}$ with $\psi\in C_c^\infty(\mathbb R)$ supported in $[-K/2,K/2]$.
Then the following are equivalent:
\begin{enumerate}
  \item[\textup{(i)}] The Riemann Hypothesis holds.
  \item[\textup{(ii)}] $Q^\star(t;\Phi)\ge0$ for every $\Phi\in\mathcal W^{\mathrm{pd}}$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Riemann Hypothesis under the corrected closure route]\label{thm:RH}
Assume the hypotheses of Theorem~\ref{thm:Main-positivity}; equivalently, assume either the scalar compact route \textup{(S1)}--\textup{(S4)} or the fallback packet route \textup{A1-pd} + packet-Rayleigh + PSD-pd + corrected local/global closure on the cone have been closed in addition to the analytic modules already established in the draft. Then the Riemann Hypothesis is true.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:Main-positivity} we have $Q^\star(t;\Phi)\ge0$ on the corrected positive-definite Weil cone $\mathcal W^{\mathrm{pd}}$ in the normalization of Section~\ref{sec:T0}. Applying Theorem~\ref{thm:Weil-criterion} yields the claim.
\end{proof}

\begin{remark}[On normalization and scope]
The normalization in \textup{(T0)} matches the Guinand--Weil conventions; thus Theorem~\ref{thm:Weil-criterion} applies to the corrected positive-definite cone. No numerical tables or ATP artifacts enter the implication from full corrected-cone positivity to RH. The extra assumptions in Theorem~\ref{thm:RH} are exactly the unresolved compact-positivity ingredients isolated in Section~\ref{sec:Main-closure}: either the scalar compact route \textup{(S1)}--\textup{(S4)} or the fallback packet package.
\end{remark}
