\subsection{Weil linkage: positivity implies the Riemann Hypothesis}\label{sec:Weil-linkage}

\begin{theorem}[Weil's positivity criterion, normalized]\label{thm:Weil-criterion}
Let $Q^\star(t;\cdot)$ be the Weil functional attached to $\zeta(s)$ in the normalization of Section~\ref{sec:T0}, and let $\mathcal W$ be the Weil cone described in Section~\ref{sec:notation}. Then the following are equivalent:
\begin{enumerate}
  \item[\textup{(i)}] The Riemann Hypothesis holds.
  \item[\textup{(ii)}] $Q^\star(t;\Phi)\ge0$ for every $\Phi\in\mathcal W$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Riemann Hypothesis under the closure gates]\label{thm:RH}
Assume the hypotheses of Theorem~\ref{thm:Main-positivity}; equivalently, assume that the unresolved closure gates \textup{(G1)}--\textup{(G5)} have been closed in addition to the analytic modules already established in the draft. Then the Riemann Hypothesis is true.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:Main-positivity} we have $Q^\star(t;\Phi)\ge0$ on the Weil cone $\mathcal W$ in the normalization of Section~\ref{sec:T0}. Applying Theorem~\ref{thm:Weil-criterion} yields the claim.
\end{proof}

\begin{remark}[On normalization and scope]
The normalization in \textup{(T0)} matches the Guinand--Weil conventions; thus Theorem~\ref{thm:Weil-criterion} applies verbatim. No numerical tables or ATP artifacts enter the implication from full Weil-class positivity to RH. The only extra assumptions in Theorem~\ref{thm:RH} are the unresolved closure gates already isolated in Section~\ref{sec:Main-closure}.
\end{remark}
