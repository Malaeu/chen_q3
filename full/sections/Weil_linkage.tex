\subsection{Weil linkage: positivity on the corrected cone implies the Riemann Hypothesis}\label{sec:Weil-linkage}

\begin{theorem}[Weil's positivity criterion, corrected target]\label{thm:Weil-criterion}
Let $Q^\star(t;\cdot)$ be the Weil functional attached to $\zeta(s)$ in the normalization of Section~\ref{sec:T0}, and let $\mathcal W^{\mathrm{pd}}$ be the positive-definite Weil cone generated locally by convolution squares as in Section~\ref{sec:scope}. Then the following are equivalent:
\begin{enumerate}
  \item[\textup{(i)}] The Riemann Hypothesis holds.
  \item[\textup{(ii)}] $Q^\star(t;\Phi)\ge0$ for every $\Phi\in\mathcal W^{\mathrm{pd}}$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Riemann Hypothesis under the corrected closure route]\label{thm:RH}
Assume the hypotheses of Theorem~\ref{thm:Main-positivity}; equivalently, assume that the centered packet density theorem `A1-pd` and the corresponding local/global closure on the corrected cone have been closed in addition to the analytic modules already established in the draft. Then the Riemann Hypothesis is true.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:Main-positivity} we have $Q^\star(t;\Phi)\ge0$ on the corrected positive-definite Weil cone $\mathcal W^{\mathrm{pd}}$ in the normalization of Section~\ref{sec:T0}. Applying Theorem~\ref{thm:Weil-criterion} yields the claim.
\end{proof}

\begin{remark}[On normalization and scope]
The normalization in \textup{(T0)} matches the Guinand--Weil conventions; thus Theorem~\ref{thm:Weil-criterion} applies to the corrected positive-definite cone. No numerical tables or ATP artifacts enter the implication from full corrected-cone positivity to RH. The only extra assumptions in Theorem~\ref{thm:RH} are the centered packet density and LF-closure steps already isolated in Section~\ref{sec:Main-closure}.
\end{remark}
