\subsection{Compact-by-compact transfer (T5, legacy)}\label{sec:T5-compact}

\noindent\textbf{Status note.} The results below capture the centered-cone part of the historical T5 route. They are kept for provenance only and are not invoked in the current mainline closure.

\subsection*{Standing analytic inputs (uniform version)}
For each $K>0$ we use the uniform analytic data from Sections~\ref{sec:A3} and~\ref{sec:rkhs-prime-cap}:
\begin{itemize}
  \item[\textbf{(A3)}] Uniform Archimedean floor: $\min_{\theta\in\TT} P_A(\theta) \ge c_*$
  (Lemma~\ref{lem:uniform-arch-floor}), which supplies an explicit $c_*>0$.
  \item[\textbf{(A3.b)}] Discretization control: for all $M\ge M_0^{\mathrm{unif}}$,
  \[
    C_{\mathrm{SB}}\,\omega_{P_A}\!\Big(\frac{1}{2M}\Big)\ \le\ \frac{c_*}{2}
  \]
  (Corollary~\ref{cor:uniform-discretisation}).
  \item[\textbf{(RKHS)}] Uniform prime contraction: for $t=t_{\mathrm{crit}}$,
  \[
    \|T_P^{\mathrm{RKHS}}(t)\|\ \le\ \rho(t)\ \le\ \frac{c_*}{4}
  \]
  (Corollary~\ref{cor:uniform-prime-cap}).
\end{itemize}

We also recall the density/continuity interface on $W_K$:
\begin{itemize}
  \item[\textbf{(A1$'$)}] The Fej\'er$\times$heat cone is dense in $W_K$.
  \item[\textbf{(A2)}] $Q^\star(t;\cdot)$ is continuous on $W_K$; specifically $|Q^\star(t;\Phi)-Q^\star(t;\Psi)|\le L_Q(K)\|\Phi-\Psi\|_\infty$.
\end{itemize}

\begin{lemma}[Grid-lift inequality]\label{lem:T5p-grid}
Assume $c_*>0$ and $t=t_{\mathrm{crit}}$. For every $K>0$ and $M\ge M_0^{\mathrm{unif}}$,
\[
  \lambda_{\min}\!\big(T_M[P_A]-T_P^{\mathrm{Ray}}(t,M)\big)
  \ \ge\ c_*\ -\ C_{\mathrm{SB}}\,\omega_{P_A}\!\Big(\tfrac{1}{2M}\Big)\ -\ \|T_P^{\mathrm{Ray}}(t,M)\|
  \ \ge\ \frac{c_*}{4}.
\]
\end{lemma}

\begin{proof}
Combine the uniform Archimedean floor $\min P_A \ge c_*$ with:
\begin{itemize}
  \item Discretization: $C_{\mathrm{SB}}\,\omega_{P_A}(1/(2M)) \le c_*/2$ for $M\ge M_0^{\mathrm{unif}}$.
  \item Prime cap: $\|T_P^{\mathrm{RKHS}}(t)\| \le \rho(t) \le c_*/4$.
\end{itemize}
Then $\lambda_{\min} \ge c_* - c_*/2 - c_*/4 = c_*/4$.
\end{proof}

\begin{theorem}[T5: legacy centered-cone transfer]\label{thm:T5-compact}
Assume $c_*>0$ and $t=t_{\mathrm{crit}}$. For every $K>0$ one has
\[
  \lambda_{\min}\!\big(T_{M}[P_A]-T_P^{\mathrm{Ray}}(t,M)\big)\ \ge\ \frac{c_*}{4}\ >\ 0
\]
for all $M\ge M_0^{\mathrm{unif}}$.
In particular, $Q^\star(t;\Phi)\ge0$ on the centered Fej\'er$\times$heat cone generated by $\Phi_{B,t,0}$ inside each $W_K$. If one additionally had density of that centered cone in $W_K$, then positivity would extend to all of $W_K$ and hence to the full Weil class.
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:T5p-grid}, the finite Toeplitz form $T_M[P_A]-T_P^{\mathrm{Ray}}(t,M)$ has positive
minimum eigenvalue $\ge c_*/4 > 0$ for all $M\ge M_0^{\mathrm{unif}}$.
Positivity on the centered Fej\'er$\times$heat cone follows.
The final extension from that centered cone to all of $W_K$ would require an additional density or compatibility statement. As noted in Remark~\ref{rem:a1-centered-obstruction}, this is exactly the step that is not available for the centered family in the present paper.
\end{proof}

\begin{remark}[No K-dependent schedules]
The uniform approach eliminates the need for K-dependent schedules $c_0(K)$, $t^\star(K)$, $M^\star(K)$.
The same uniform parameters $c_*$, $t=t_{\mathrm{crit}}$,
and $M_0^{\mathrm{unif}}$ work for all compacts $[-K,K]$ simultaneously within this legacy centered-cone skeleton.
\end{remark}
