\begin{theorem}[AB$\boldsymbol{\infty}$ legacy centered-cone closure (fixed $q=30$, BRC--SAFE default)]\label{thm:ABinfty}
Work under the T0 normalisation $Q^\star(t;\cdot)=Q_{\mathrm{GW}}$ on the GW-axis.
Fix $q_0=30$ and the single heat scale $t=t_{\mathrm{crit}}$. Let $\{K_i\}_{i\ge1}$ be an increasing chain with $\bigcup_i[-K_i,K_i]=\mathbb{R}$.
For each $i$ choose parameters $(B_i,M_i)$ at the fixed scale $t=t_{\mathrm{crit}}$, and work with the unshifted node $\tau=0$ (so the shift grid is $\{0\}$).
Assume for every $i$:

\begin{description}[leftmargin=3.2em,labelwidth=2em]
\item[\textbf{(R)} Arch floor (A3).]
With the centered Fej\'er$\times$heat window $\Phi_{B_i,t,0}$ and Arch symbol $P_A$,
\begin{equation}\label{eq:ab_infinity_lemma-formula}
\min_{\theta} P_A(\theta)\ \ge\ c_0(K_i).
\end{equation}

\item[\textbf{(N)} Nyquist \& Norm.]
The symbol modulus and the prime cap satisfy
\begin{equation}\label{eq:ab_infinity_lemma-omega}
C\,\omega_{P_A}\!\Bigl(\tfrac{1}{2M_i}\Bigr)\ \le\ \tfrac{c_0(K_i)}{2},
\qquad
\|T_P^{(q_0)}(t_{\mathrm{crit}})\|\ \le\ \tfrac{c_0(K_i)}{2},
\end{equation}
where $T_P^{(q_0)}(t_{\mathrm{crit}})$ is the modular cap at modulus $q_0=30$ with RKHS smoothing scale $t=t_{\mathrm{crit}}$.

\item[\textbf{(A)} Grid$\to$continuum (BRC--SAFE default).]
The (degenerate) grid at $\tau=0$ is \textup{\textsc{BRC}}--\textup{\textsc{SAFE}}; equivalently, the resolvent
certificate with Ky Fan/Hoffman--Wielandt budget holds at the unshifted node.
\end{description}

Then $Q^\star(t;\Phi)\ge0$ for all tests $\Phi$ in the centered legacy cone generated by the unshifted atoms $\Phi_{B_i,t,0}$ on $[-K_i,K_i]$, for every $i$.
Any passage from this centered legacy cone to the full Weil class additionally requires a density or compatibility hypothesis not supplied by the present appendix.
\end{theorem}

\begin{proof}[Proof (by plumbing)]
By the Toeplitz symbol bridge (A3), at the unshifted node $\tau=0$,
\begin{equation}\label{eq:ab_infinity_lemma-lambda-min}
\lambda_{\min}\!\big(T_{M_i}[P_A]-T_P^{\mathrm{Ray}}(t,M_i)\big)
\ \ge\ \min P_A\ -\ C\,\omega_{P_A}\!\Bigl(\tfrac{1}{2M_i}\Bigr)\ -\ \|T_P^{\mathrm{Ray}}(t,M_i)\|.
\end{equation}
Assumptions \textbf{(R)}--\textbf{(N)} make the RHS $\ge c_0 - \tfrac{c_0}{2} - \tfrac{c_0}{2}=0$, so nonnegativity holds on all grid nodes.
By \textbf{(A)} (\textup{\textsc{BRC}}--\textup{\textsc{SAFE}} at $\tau=0$) the sign is preserved on $[-K_i,K_i]$.
This yields nonnegativity on the centered legacy cone generated by the unshifted Fej\'er$\times$heat windows.
The historical jump from that cone to all tests on $[-K_i,K_i]$ is no longer part of the current proof, because the centered family is not known to be dense.
\end{proof}

\begin{remark}[Lipschitz-lift option]
Instead of \textup{\textsc{BRC}}--\textup{\textsc{SAFE}} one may enforce the deterministic Lipschitz lift
$L_Q(K_i)\,L_\Phi(K_i)\,\Delta\tau \le c_0(K_i)/4$; in the unshifted case $\Delta\tau=0$ so the condition is immediate.
\end{remark}

\begin{remark}[Monotone inheritance]
It is convenient (not essential) to choose $B_i\uparrow$, $M_i\uparrow$, and nonincreasing budgets so that acceptance persists along the chain.
\end{remark}
