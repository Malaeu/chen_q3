% Theorem A1' --- Auxiliary density on the broad restriction cone

This section is now \emph{auxiliary} for the RH mainline. It records the shifted
density theorem on the broad restriction cone
\[
  R_K := C^+_{\mathrm{even}}([-K,K])
\]
with the uniform norm $\|\cdot\|_\infty$. After the target-cone audit of
Section~\ref{sec:scope}, this theorem no longer serves as the public density
statement feeding RH; the corrected missing theorem is the centered positive-definite
density statement A1-pd. For the corrected route we fix
\[
  \widetilde{\psi}(x):=\overline{\psi(-x)},
  \qquad
  \mathcal W_{K,0}^{\mathrm{pd}}
  :=\bigl\{\psi*\widetilde{\psi}:\ \psi\in C_c^\infty(\mathbb R),\ \operatorname{supp}\psi\subset[-K/2,K/2]\bigr\},
\]
\[
  \mathcal W_K^{\mathrm{pd}}
  :=\overline{\operatorname{cone}\!\bigl(\mathcal W_{K,0}^{\mathrm{pd}}\bigr)}^{\|\cdot\|_\infty}
  \subset C_c([-K,K])_{\mathrm{even}}.
\]

\begin{theorem}[A1' --- auxiliary density on $R_K$]\label{thm:A1-density}
For every compact $[-K,K]$ the cone $\{\text{Fej\'er}*\text{heat approximants}\}$ is dense in $R_K$ in $\|\cdot\|_\infty$.
\end{theorem}
\begin{proof}
This is an immediate corollary of Theorem~\ref{a1:thm:A1-local-density}, which provides the explicit uniform approximation on $[-K,K]$ by finite nonnegative combinations of Fej\'er$\times$heat atoms.
\end{proof}

\begin{remark}[PW reinforcement]
On $[-K,K]$ the heat kernel satisfies $\widehat{\rho_t}(s)=e^{-4\pi^2 t s^2}\ge e^{-4\pi^2 t K^2}>0$, hence the convolution with $\rho_t$ is invertible on the compact in the PW metric. Together with the Fej\'er (positive) hat interpolation and a Weierstrass/Fej\'er--Riesz approximation step in $\|\cdot\|_{\infty}$, this yields cone density in the broad restriction setting. These constants are still useful as auxiliary approximation data, but they do not by themselves identify the corrected positive-definite Weil cone.
\end{remark}

\begin{definition}[Pre-square packet span and density cone]\label{def:pd-packet-cone}
Fix $t_0>0$. For packet atoms
\[
  g_{\delta,\tau}(\xi):=\Lambda_\delta(\xi-\tau)\rho_{t_0}(\xi-\tau),
\]
let
\[
  \Psi(\xi)=\sum_{j=1}^N c_j\,g_{\delta_j,\tau_j}(\xi)
\]
be a finite Fej\'er$\times$heat pre-packet with $\operatorname{supp}\Psi\subset[-K/2,K/2]$.
Set
\[
  \mathcal P_K(t_0)
  := \operatorname{span}\!\Bigl\{
      g_{\delta,\tau}:\ |\tau|+\delta\le K/2
    \Bigr\},
  \qquad
  \Phi_\Psi:=\Psi*\widetilde{\Psi}.
\]
The centered density cone produced by this pre-square packet space is
\[
  \mathcal G_K^{\mathrm{pd}}
  := \operatorname{cone}\!\Bigl\{
      \Phi_\Psi:\ \Psi\in \mathcal P_K(t_0)
    \Bigr\}.
\]
Then $\mathcal G_K^{\mathrm{pd}}\subseteq \mathcal W_K^{\mathrm{pd}}$ by construction.
\end{definition}

\begin{theorem}[Target theorem A1-pd: centered packet density in the corrected cone]\label{thm:A1-pd}
For every $K>0$,
\[
  \overline{\mathcal G_K^{\mathrm{pd}}}^{\|\cdot\|_\infty}
  =
  \mathcal W_K^{\mathrm{pd}}.
\]
\end{theorem}
\begin{proof}[Proof sketch]
The proof of Lemma~6.3 already yields uniform approximation on compact windows by
finite linear combinations of shifted Fej\'er$\times$heat atoms. This gives
density of $\mathcal P_K(t_0)$ in $C_c([-K/2,K/2])$. Since the autocorrelation map
$\Psi\mapsto \Psi * \widetilde{\Psi}$ is continuous from $L^1$ to $L^\infty$ by
Lemma~\ref{lem:autocorrelation-continuity}, the induced cone of autocorrelation
packets is dense in the corrected positive-definite cone.
\end{proof}

\begin{remark}[Why A1-pd is the right density statement]\label{rem:A1-pd-right-target}
The theorem A1-pd differs essentially from A1$'$. The shifted theorem A1$'$ proves
density in the broad restriction cone $R_K=C^+_{\mathrm{even}}([-K,K])$, whereas
Theorem~\ref{thm:A1-pd} asks for density in the positive-definite cone generated by
convolution squares. The density family is therefore centered only after passing to
autocorrelations: the shift parameters live inside the pre-square packet $\Psi$, not
as free shifts of the final test $\Phi_\Psi$. This is exactly why A1-pd survives the
target-cone audit while the broad-cone density route does not.
\end{remark}

\begin{lemma}[Autocorrelation continuity in the pre-square variable]\label{lem:autocorrelation-continuity}
Let $\psi,\varphi\in L^1(\mathbb R)$. Then
\[
  \|\psi*\widetilde{\psi}-\varphi*\widetilde{\varphi}\|_\infty
  \le
  \bigl(\|\psi\|_{L^1}+\|\varphi\|_{L^1}\bigr)\,\|\psi-\varphi\|_{L^1}.
\]
\end{lemma}

\begin{proof}
Write
\[
  \psi*\widetilde{\psi}-\varphi*\widetilde{\varphi}
  =
  (\psi-\varphi)*\widetilde{\psi}
  + \varphi*(\widetilde{\psi}-\widetilde{\varphi}).
\]
Young's inequality gives
\[
  \|(\psi-\varphi)*\widetilde{\psi}\|_\infty
  \le \|\psi-\varphi\|_{L^1}\,\|\widetilde{\psi}\|_{L^1}
  = \|\psi-\varphi\|_{L^1}\,\|\psi\|_{L^1},
\]
and similarly
\[
  \|\varphi*(\widetilde{\psi}-\widetilde{\varphi})\|_\infty
  \le \|\varphi\|_{L^1}\,\|\widetilde{\psi}-\widetilde{\varphi}\|_{L^1}
  = \|\varphi\|_{L^1}\,\|\psi-\varphi\|_{L^1}.
\]
Adding the two bounds gives the claim.
\end{proof}

\begin{remark}[Proof skeleton for A1-pd]\label{rem:A1-pd-skeleton}
The intended proof of Theorem~\ref{thm:A1-pd} has two layers:
\begin{enumerate}
  \item prove packet density in the pre-square space
  $C_c^\infty([-K/2,K/2])$ in the $L^1$ topology, using the flexibility of
  shifted Fej\'er$\times$heat packets as the approximation engine;
  \item pass from $\psi$ to $\psi*\widetilde{\psi}$ via
  Lemma~\ref{lem:autocorrelation-continuity}.
\end{enumerate}
This is the corrected version of the old A1 technology: the approximation happens
at the level of the packet $\Psi$, while the final test $\Phi_\Psi$ remains inside
the positive-definite cone.
\end{remark}

\begin{remark}[What A1-pd does \emph{not} yet prove]\label{rem:a1-pd-not-same-family}
Theorem~\ref{thm:A1-pd} produces density for the cone
\[
  \mathcal G_K^{\mathrm{pd}}
  = \operatorname{cone}\{\Psi*\widetilde{\Psi}:\ \Psi\in\mathcal P_K(t_0)\}.
\]
This theorem by itself does not yet yield positivity on the corrected cone. What it
does provide is the exact dense packet family for the corrected route. The naive
centered family $\Phi_{B,t}|p|^2$ is mathematically natural, but it is too large to
serve as that exact family: combined with the full quadratic-form meaning of
Lemma~8.8 and the centered A3 positivity engine, it would overgenerate false broad
local positivity on compact bumps. The honest corrected package is therefore:
\begin{enumerate}
  \item Theorem~\ref{thm:A1-pd} on the dense family $\mathcal G_K^{\mathrm{pd}}$;
  \item the exact packet-Rayleigh theorem on autocorrelation packets
  $\Psi_c * \widetilde{\Psi_c}$;
  \item the new hard theorem A3-pd, asserting positivity of the packet symbol
  $S_{g,\Delta}$ on that same exact family.
\end{enumerate}
\end{remark}

\begin{lemma}[Compact support convolution reduction]\label{lem:convolution-compact-support}
Let $f\in C_c(\mathbb{R})$ be compactly supported with $\mathrm{supp}(f)\subseteq[-L,L]$, and let $g:\mathbb{R}\to\mathbb{R}$ be any function. Then for all $x\in\mathbb{R}$,
\[
(f*g)(x) = \int_{\mathbb{R}} f(y)\,g(x-y)\,dy = \int_{-L}^{L} f(y)\,g(x-y)\,dy.
\]
\end{lemma}
\begin{proof}
Since $f(y)=0$ for $|y|>L$, the integrand vanishes outside $[-L,L]$.
\end{proof}

\begin{theorem}[A1' (density via shifted evenized atoms)]\label{a1:thm:A1-local-density}
Let $K=[-R,R]$ with $R>0$. Fix a heat scale $t>0$ and, for $B>0$ and $\tau\in[-R,R]$, define the even nonnegative frequency window
\[
  \Phi_{B,t,\tau}(\xi)
  := \frac12\Bigl[\Lambda_B(\xi-\tau)\,\rho_t(\xi-\tau)+\Lambda_B(\xi+\tau)\,\rho_t(\xi+\tau)\Bigr],
\]
where $\Lambda_B(x)=(1-|x|/B)_+$ and $\rho_t(x)=(4\pi t)^{-1/2} e^{-x^2/(4t)}$ (so $\int_{\mathbb R}\rho_t=1$, $\rho_t\ge0$). Let $\mathcal C$ be the closed convex cone generated by finite nonnegative combinations of $\{\Phi_{B,t,\tau}\}$ with $B$ sufficiently large (depending on $R$) and $\tau\in[-R,R]$. Then $\mathcal C$ is dense in
\[
  R_K = C^+_{\mathrm{even}}([-R,R])
\]
in the uniform norm.
\end{theorem}
\begin{proof}
This is exactly the shifted-atom density theorem proved in Appendix~\ref{app:legacy-a1-density}; we restate it here because it remains the correct auxiliary density statement on $C^+_{\mathrm{even}}([-R,R])$.
\end{proof}

\begin{remark}[Parameter scaling in A1']\label{rem:a1-scaling}
The bandwidth $B$ required in Theorem~\ref{a1:thm:A1-local-density}
depends on the compact $[-R,R]$ and the target accuracy $\varepsilon$,
while the heat scale $t$ may be held fixed throughout the approximation. Accordingly, only the
bandwidth and the shift grid vary with the compact and the target accuracy; no K-dependent heat schedule is required.
\end{remark}

\begin{remark}[Centered subcone is not dense in the broad restriction cone]\label{rem:a1-centered-obstruction}
If one freezes $\tau=0$, then each generator $\Phi_{B,t,0}$ is even and nonincreasing on $[0,\infty)$.
Finite nonnegative combinations and uniform limits preserve this monotonicity. Consequently the
centered subcone cannot be dense in $R_K=C^+_{\mathrm{even}}([-R,R])$; for example, the even nonnegative
function $\xi\mapsto 1+\xi^2$ on $[-R,R]$ is not nonincreasing on $[0,R]$. This obstruction is about
the broad cone $R_K$ only. It does \emph{not} rule out a centered density theorem on the corrected
positive-definite cone $\mathcal W_K^{\mathrm{pd}}$, where the target geometry is different.
\end{remark}

\begin{remark}[What A1' does and does not provide now]\label{rem:a1-support-gap}
Theorem~\ref{a1:thm:A1-local-density} remains a valuable density theorem on the broad restriction cone $R_K$.
What it does \emph{not} now provide is the mainline RH density theorem. After the target-cone audit, the missing theorem is A1-pd: density of centered Fej\'er$\times$heat / autocorrelation packets in the corrected positive-definite cone $\mathcal W_K^{\mathrm{pd}}$.
\end{remark}
