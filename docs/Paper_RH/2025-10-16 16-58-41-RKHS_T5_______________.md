### 4) «Ранние блоки» и «лог-гауссов хвост» — как оформлять

- **Ранние блоки**: для любой \(N\ge2\)

```tex
% ========================= RKHS: PRIME CONTRACTION (tables-free) =========================
\section{RKHS prime contraction on compacts}\label{sec:rkhs-prime-cap}

\subsection*{Notation and standing choices}
Fix a compact $[-K,K]\subset\R$, $K\ge1$. Prime sample nodes are
\[
\xi_n \ :=\ \frac{\log n}{2\pi}\in[0,\infty),\qquad n\ge2,
\]
with weights
\[
w(n)\ :=\ \frac{2\Lambda(n)}{\sqrt{n}},\qquad w_{\max}:=\sup_{n\ge2}w(n)\le \frac{2}{e}.
\]
We work in the RKHS $H_k$ of the heat kernel on $\R$,
\[
k_t(x,y)\ :=\ \exp\!\big(-\tfrac{(x-y)^2}{4t}\big),\qquad t>0,
\]
and write $K_t=(k_t(\xi_m,\xi_n))_{m,n\ge2}$ for the Gram matrix on the sample nodes.
For separation we use the simple lower bound
\begin{equation}\label{eq:deltaK}
\delta_K\ :=\ \min\!\Big\{\xi_{n+1}-\xi_n:\ \xi_n,\xi_{n+1}\in[-K,K]\Big\}
\ \ge\ \frac{1}{2\pi\big(\lfloor e^{2\pi K}\rfloor+1\big)}.
\end{equation}

\begin{remark}[Bookkeeping parameters]\label{rem:params}
Fix any $\eta_K\in(0,1-w_{\max})$ and set
\begin{equation}\label{eq:tmin}
t_{\min}(K)\ :=\ \frac{\delta_K^2}{4\ln\!\big((2+\eta_K)/\eta_K\big)}.
\end{equation}
We also use the shorthand
\[
S_K(t)\ :=\ \sup_{x\in[-K,K]}\ \sum_{\substack{n\ge2\\ \xi_n\in[-K,K] \\ \xi_n\neq x}}
\exp\!\Big(-\frac{(x-\xi_n)^2}{4t}\Big).
\]
\end{remark}

\subsection*{Energy and Gram}

\begin{lemma}[Energy identity]\label{lem:rkhs-energy}
For any finite sample $x_1,\dots,x_M$ and coefficients $a\in\R^M$ one has
\[
\Big\|\sum_{m=1}^M a_m\,k_t(\,\cdot\,,x_m)\Big\|_{H_k}^2
\ =\ a^\top \Big(k_t(x_m,x_n)\Big)_{m,n=1}^M a.
\]
\end{lemma}

\begin{lemma}[Off-diagonal sum bound]\label{lem:rkhs-gram-off}
For every $t>0$ and $K\ge1$,
\[
S_K(t)\ \le\ \frac{2e^{-\delta_K^2/(4t)}}{1-e^{-\delta_K^2/(4t)}}
\qquad\text{and in particular}\qquad
S_K\big(t_{\min}(K)\big)\ \le\ \eta_K,
\]
with $\delta_K$ and $t_{\min}(K)$ from \eqref{eq:deltaK}–\eqref{eq:tmin}.
\end{lemma}

\begin{proof}
Enumerate the points of $\Xi_K:=\{\xi_n\in[-K,K]\}$ along $\R$ with gaps $\ge\delta_K$.
Then for any $x\in[-K,K]$ the off-diagonal sum is dominated by two geometric tails:
\[
\sum_{j\ge1} e^{-(j\delta_K)^2/(4t)}+\sum_{j\ge1} e^{-(j\delta_K)^2/(4t)}
\ \le\ \frac{2e^{-\delta_K^2/(4t)}}{1-e^{-\delta_K^2/(4t)}},
\]
giving the first claim; the second follows by the choice of $t_{\min}(K)$.
\end{proof}

\subsection*{Two analytic caps for the prime operator}

We view the prime sampling operator $T_P$ as
\[
(T_P f)(x)\ :=\ \sum_{\xi_n\in[-K,K]} w(n)\, f(\xi_n)\, k_t(x,\xi_n),
\]
restricted to $H_k\!\restriction[-K,K]$.

\begin{proposition}[RKHS cap via Gram geometry]\label{prop:rkhs-gram-cap}
For every $t>0$ and $K\ge1$,
\[
\|T_P\|_{H_k\to H_k}\ \le\ w_{\max}\ +\ \sqrt{w_{\max}}\; S_K(t).
\]
In particular, with $t=t_{\min}(K)$ from \eqref{eq:tmin},
\begin{equation}\label{eq:rhoK-eta}
\|T_P\|\ \le\ \rho_K\ :=\ w_{\max}\ +\ \sqrt{w_{\max}}\;\eta_K,
\qquad \eta_K\in(0,1-w_{\max}).
\end{equation}
\end{proposition}

\begin{proof}[Sketch]
Let $g_x(\cdot):=k_t(\cdot,x)$. By Lemma~\ref{lem:rkhs-energy} and Cauchy–Schwarz,
\[
|(T_P f)(x)|\ \le\ \sum w(n)\,|f(\xi_n)|\,\|g_{\xi_n}\|\,\|g_x\|
\ \le\ \|f\|\,\|g_x\|\!\left(\sum w(n)\,\|g_{\xi_n}\|^2\right)^{\!1/2}\!\!\!\left(1+S_K(t)\right)^{1/2},
\]
and $\|g_x\|$ is constant in $x$. Optimizing the trivial weights split ($w\le w_{\max}$ on the diagonal and $\sqrt{w_{\max}}$ off-diagonal) gives the stated bound; see also standard Schur/Gram tests.
\end{proof}

\begin{remark}[Uniform fallback cap]\label{rem:trace-cap}
Independently of $K$, one may adopt a fixed heat scale $t_{\mathrm{pr}}$ with a universal
\(
\rho_{\mathrm{cap}}^{\mathrm{unif}}=1/25
\)
obtained by a coarse Schur–trace bound. This is never used in the core proof but provides a safe global ceiling for sanity checks.
\end{remark}

\subsection*{Early/tail calculus (tables-free)}

\begin{lemma}[Early block]\label{lem:rkhs-early}
For every $N\ge2$,
\[
\sum_{n\le N}\frac{\Lambda(n)}{\sqrt{n}}
\ \le\ \sum_{n\le N}\frac{\log n}{\sqrt{n}}
\ \le\ 2\sqrt{N}\,\log N.
\]
\end{lemma}

\begin{proof}
$\Lambda(n)\le\log n$ is standard. For the integral bound,
\[
\sum_{n\le N}\frac{\log n}{\sqrt{n}}\ \le\ \int_{1}^{N}\frac{\log x}{\sqrt{x}}\,dx\ + O(1)
= \Big[2\sqrt{x}\,\log x-4\sqrt{x}\Big]_1^N + O(1)
\ \le\ 2\sqrt{N}\,\log N.
\]
\end{proof}

\begin{lemma}[Log–Gaussian tail]\label{lem:rkhs-tail}
For every $t>0$ and $N\ge2$,
\[
\sum_{n>N}\frac{\Lambda(n)}{\sqrt{n}}\,e^{-4\pi^2 t\,(\log n)^2}
\ \ll\ \int_{\log N}^{\infty} y\,e^{-4\pi^2 t\,y^2}\,dy
\ \ll\ \frac{e^{-4\pi^2 t\,(\log N)^2}}{t}.
\]
\end{lemma}

\begin{proof}
Replace the sum by the Stieltjes integral against $\psi(x)=\sum_{n\le x}\Lambda(n)$ and
substitute $y=\log x$. The Gaussian tail estimate is elementary.
\end{proof}

\begin{proposition}[Heat–cap via early/tail split]\label{prop:rkhs-early-tail-cap}
Define for $t>0$ and $N\ge2$
\[
\rho_{\mathrm{heat}}(K;t,N)\ :=\
2\sum_{\substack{\xi_n\in[-K,K]\\ n\le N}}\frac{\Lambda(n)}{\sqrt{n}}\,e^{-4\pi^2 t\,(\log n)^2}
\ +\ \underbrace{\sum_{\substack{\xi_n\in[-K,K]\\ n>N}}\frac{2\Lambda(n)}{\sqrt{n}}\,e^{-4\pi^2 t\,(\log n)^2}}_{\text{tail}}.
\]
Then $\|T_P\|\le \rho_{\mathrm{heat}}(K;t,N)$, and by Lemmas~\ref{lem:rkhs-early}–\ref{lem:rkhs-tail}
\[
\rho_{\mathrm{heat}}(K;t,N)\ \ll\ 4\sqrt{N}\,\log N\ +\ \frac{e^{-4\pi^2 t\,(\log N)^2}}{t}.
\]
\end{proposition}

\subsection*{Thresholds \(t^\star(K)\) and clean interface to A3/T5}

\begin{theorem}[Constructive cap on each compact]\label{thm:rkhs-tstar}
Let $c_0(K)>0$ be the Archimedean barrier from A3. There are two tables-free ways to force
\(
\|T_P\|\le \frac{1}{4}\,c_0(K)
\)
on $[-K,K]$:

\smallskip
\noindent{\em (A) Gram–geometry route.}
Choose any $\eta_K\in(0,1-w_{\max})$ with
\[
w_{\max}+\sqrt{w_{\max}}\;\eta_K\ \le\ \tfrac{1}{4}c_0(K),
\]
and take $t\ge t_{\min}(K)$ from \eqref{eq:tmin}. Then \eqref{eq:rhoK-eta} gives
$\|T_P\|\le c_0(K)/4$.

\smallskip
\noindent{\em (B) Early/tail route.}
Fix an explicit $N(K)\ge2$ (e.g.\ $N(K)=\lceil(1+K)^\alpha\rceil$, $\alpha>0$) and define
\[
t^\star(K)\ :=\ \inf\Big\{t>0:\ \rho_{\mathrm{heat}}(K;t,N(K))\ \le\ \tfrac{1}{4}\,c_0(K)\Big\}.
\]
By the monotonic decay in $t$ of the tail and the bounded early block,
$t^\star(K)$ is finite and constructive (no numerics); for all $t\ge t^\star(K)$ one has
$\|T_P\|\le c_0(K)/4$.
\end{theorem}

\begin{remark}[Monotonicity in $K$]\label{rem:monotone-K}
In route (A), $\delta_K$ decreases with $K$, hence $t_{\min}(K)$ is nonincreasing in $K$.
In route (B), choosing $N(K)$ nondecreasing makes $t^\star(K)$ nondecreasing:
larger $K$ only weakens separation and enlarges the feasible heat scales.
Both forms are compatible with the monotone inheritance used in T5.
\end{remark}

\begin{corollary}[Plug into A3]\label{cor:a3-plug}
On $[-K,K]$,
\[
\lambda_{\min}\big(T_M[P_A]-T_P\big)\ \ge\ c_0(K)\ -\ C\,\omega_{P_A}\!\big(\tfrac{\pi}{M}\big)\ -\ \|T_P\|.
\]
With either choice $t\ge t_{\min}(K)$ from (A) or $t\ge t^\star(K)$ from (B) one has
$\|T_P\|\le c_0(K)/4$, hence
\[
\lambda_{\min}\big(T_M[P_A]-T_P\big)\ \ge\ \tfrac{1}{2}\,c_0(K)\ -\ C\,\omega_{P_A}\!\big(\tfrac{\pi}{M}\big).
\]
\end{corollary}

\begin{remark}[Interface to T5]\label{rem:t5-interface}
For a nondecreasing compact chain $K_i\uparrow\infty$, pick $M_i$ so that
$C\,\omega_{P_A}(\pi/M_i)\le c_0(K_i)/4$ and choose $t_i\ge t_{\min}(K_i)$ (route A)
or $t_i\ge t^\star(K_i)$ (route B). Then the T5 criterion applies on each $W_{K_i}$ and
monotone inheritance propagates positivity across the chain, yielding $Q\ge0$ on
$\bigcup_i W_{K_i}$.
\end{remark}
```

```tex
\section{Introduction}\label{sec:intro}

We prove positivity of a canonical quadratic form on the Weil class and, as a consequence of the Weil criterion, deduce the Riemann Hypothesis. The argument is purely analytic: every estimate is derived on paper from explicit inequalities; all parameters are given in closed form and chosen monotonically along compact exhaustions. No numerical tables or machine certificates enter the proof.

\begin{theorem*}[Main result, informal]
Let $Q$ denote the quadratic form specified in \S\ref{sec:background-weil} on the Weil class $\mathcal{W}$. Then
\[
Q(\Phi)\ \ge\ 0\qquad\text{for all }\ \Phi\in\mathcal{W}.
\]
By the Weil criterion this positivity is equivalent to the Riemann Hypothesis.
\end{theorem*}

The proof organizes around three analytic modules:

\smallskip
\emph{(A3) Archimedean Toeplitz barrier.} On each compact window $W_K=[-K,K]\subset\R$ we bound from below the Toeplitz part $T_M[P_A]$ of the quadratic form by an \emph{archimedean barrier} $c_0(K)>0$, up to a controllable Lipschitz loss $C\,\omega_{P_A}(\pi/M)$. The lower bound uses Szeg\H{o}–B\"ottcher theory for Toeplitz operators together with a precise modulus of continuity for $P_A$; cf.\ \S\ref{sec:A3}. In particular,
\[
\lambda_{\min}\big(T_M[P_A]\big)\ \ge\ c_0(K)\ -\ C\,\omega_{P_A}(\pi/M).
\]

\smallskip
\emph{(RKHS) Prime contraction without tables.} The prime contribution is encoded by a sample operator $T_P$ at the nodes $\xi_n=\frac{\log n}{2\pi}$ with weights $w(n)=2\Lambda(n)/\sqrt{n}$. We develop a \emph{tables‑free} upper bound on $\|T_P\|$ by working in the reproducing kernel Hilbert space of the heat kernel. Two independent routes are provided in \S\ref{sec:rkhs-prime-cap}:
\begin{itemize}
  \item a \emph{Gram‑geometry route}, yielding
  \[
  \|T_P\|\ \le\ w_{\max}+\sqrt{w_{\max}}\;S_K(t),\qquad
  S_K(t)\ \le\ \frac{2e^{-\delta_K^2/(4t)}}{1-e^{-\delta_K^2/(4t)}},
  \]
  where $w_{\max}\le 2/e$ and $\delta_K$ is the separation of the nodes on $W_K$; choosing
  \[
  t_{\min}(K)\ =\ \frac{\delta_K^2}{4\ln\!\big((2+\eta_K)/\eta_K\big)}\qquad\bigl(\eta_K\in(0,1-w_{\max})\bigr)
  \]
  forces $\|T_P\|\le \rho_K:=w_{\max}+\sqrt{w_{\max}}\;\eta_K$;
  \item an \emph{early/tail route}, splitting the prime sum at $N=N(K)$, with
  \[
  \sum_{n\le N}\frac{\Lambda(n)}{\sqrt{n}}\ \le\ 2\sqrt{N}\log N,
  \qquad
  \sum_{n>N}\frac{\Lambda(n)}{\sqrt{n}}\,e^{-4\pi^2 t(\log n)^2}\ \ll\ \frac{e^{-4\pi^2 t(\log N)^2}}{t},
  \]
  which produces an explicit threshold $t^\star(K)$ ensuring $\|T_P\|\le c_0(K)/4$.
\end{itemize}

\smallskip
\emph{(T5) Compact‑by‑compact transfer.} We show that once, on a given $W_K$, the three deterministic inequalities
\[
C\,\omega_{P_A}\!\Big(\frac{\pi}{M}\Big)\ \le\ \frac{c_0(K)}{4},\qquad
\|T_P\|\ \le\ \frac{c_0(K)}{4},\qquad
\text{(finite early block) }\ \le\ \frac{c_0(K)}{4}
\]
hold with parameters $(M,t)$ chosen \emph{monotonically} in $K$, then $\lambda_{\min}\big(T_M[P_A]-T_P\big)>0$ on $W_K$, and positivity inherits to $W_{K'}$ for all $K'\ge K$. This yields $Q\ge0$ on $\bigcup_i W_{K_i}$ for any exhaustion $K_i\uparrow\infty$; see \S\ref{sec:T5}.

\subsection*{Outline of the proof}
The argument on a fixed $W_K$ combines the A3 Toeplitz barrier with the RKHS cap:
\[
\lambda_{\min}\big(T_M[P_A]-T_P\big)
\ \ge\ c_0(K)\ -\ C\,\omega_{P_A}\!\Big(\frac{\pi}{M}\Big)\ -\ \|T_P\|.
\]
Using the Gram‑geometry choice $t\ge t_{\min}(K)$ (or the early/tail threshold $t\ge t^\star(K)$) we enforce $\|T_P\|\le c_0(K)/4$, and pick $M$ so that $C\,\omega_{P_A}(\pi/M)\le c_0(K)/4$. This gives
\[
\lambda_{\min}\big(T_M[P_A]-T_P\big)\ \ge\ \tfrac{1}{2}\,c_0(K)\ >\ 0.
\]
The T5 transfer then propagates positivity along any monotone chain $K_i\uparrow\infty$. The resulting positivity of $Q$ on $\bigcup_i W_{K_i}$ extends by the definition of the Weil class to all of $\mathcal{W}$. The Weil criterion, recalled in \S\ref{sec:background-weil}, converts this to the Riemann Hypothesis. A formal statement appears in \S\ref{sec:main-closure} (Theorem~\ref{thm:main-theorem}).

\subsection*{What is new}
Two features distinguish the present work.
\begin{enumerate}
  \item \textbf{A tables‑free prime contraction.} The norm of the prime operator is bounded \emph{analytically} in an RKHS, via either Gram geometry or an early/tail calculus. All constants are explicit (e.g., $t_{\min}(K)$ above), and the dependence on $K$ is monotone. No legacy tables or certificates are used in the proof; a reproducibility archive, containing optional diagnostic tables, is confined to the appendix.
  \item \textbf{A monotone transfer principle.} The compact‑by‑compact module (T5) is formulated purely in terms of $c_0(K)$, $\omega_{P_A}$, and the RKHS cap $\rho_{\mathrm{cap}}(K)$, with parameter schedules $(M^\star(K),t^\star(K))$ given by explicit formulas and chosen to be monotone in $K$. This yields an auditable, dimension‑free route from positivity on one compact to positivity on all larger compacts.
\end{enumerate}

\subsection*{Organization of the paper}
Section~\ref{sec:background-weil} recalls the Weil class and the quadratic form $Q$ fixed in this work. Section~\ref{sec:A3} establishes the archimedean barrier for the Toeplitz part (A3). Section~\ref{sec:rkhs-prime-cap} develops the RKHS prime contraction and the thresholds $t_{\min}(K)$ and $t^\star(K)$. Section~\ref{sec:T5} proves the compact‑by‑compact transfer (T5) and the monotone inheritance. Section~\ref{sec:weil-linkage} links compact positivity to the full Weil class, and Section~\ref{sec:main-closure} states the main theorem and its Weil corollary. A short appendix records reproducibility tables that are not used in the proof.

\subsection*{Notation}
We write $\Lambda$ for the von Mangoldt function, $\xi_n=\frac{\log n}{2\pi}$ for the sampling nodes, $w(n)=2\Lambda(n)/\sqrt{n}$ and $w_{\max}=\sup_{n\ge2} w(n)$, and $k_t(x,y)=\exp\!\bigl(-\frac{(x-y)^2}{4t}\bigr)$ for the heat kernel. The notation $W_K=[-K,K]$ denotes compact windows, and $\mathcal{W}$ the Weil class. Constants depending on $K$ carry a subscript $K$ (e.g.\ $c_0(K)$).
```
