/-
A sequence `κ` is a moment sequence if it is the Fourier coefficients of a finite measure on the circle.
The Toeplitz matrix of `κ` is the matrix with entries `κ(i-j)`.
-/
open Complex MeasureTheory Matrix

def IsMomentSequence (κ : ℤ → ℂ) : Prop :=
  ∃ (μ : Measure Circle), IsFiniteMeasure μ ∧
    ∀ (n : ℤ), κ n = ∫ (z : Circle), ((z ^ n : Circle) : ℂ) ∂μ

def ToeplitzMatrix (κ : ℤ → ℂ) (n : ℕ) : Matrix (Fin n) (Fin n) ℂ :=
  fun i j => κ (i - j)

/-
We define a trigonometric polynomial associated with a vector `c`.
The theorem states that the quadratic form of the Toeplitz matrix of a moment sequence `κ` with respect to `c` is equal to the integral of the squared norm of the trigonometric polynomial associated with `c`.
-/
open Complex MeasureTheory Matrix

def trigonometricPolynomial {n : ℕ} (c : Fin n → ℂ) (z : Circle) : ℂ :=
  ∑ i, c i * ((z ^ (-(i : ℤ)) : Circle) : ℂ)

theorem toeplitz_quadratic_form_eq_integral_norm_sq
    {κ : ℤ → ℂ} {μ : Measure Circle} [IsFiniteMeasure μ]
    (hκ : ∀ (n : ℤ), κ n = ∫ (z : Circle), ((z ^ n : Circle) : ℂ) ∂μ)
    (n : ℕ) (c : Fin n → ℂ) :
    dotProduct (star c) ((ToeplitzMatrix κ n).mulVec c) =
      ∫ (z : Circle), ((Complex.normSq (trigonometricPolynomial c z) : ℝ) : ℂ) ∂μ := by
        -- Expand the quadratic form `c^* T c` as `∑ i j, conj (c i) * κ (i - j) * c j`.
        have h_quad_form : Star.star c ⬝ᵥ (ToeplitzMatrix κ n) *ᵥ c = ∑ i : Fin n, ∑ j : Fin n, (star (c i)) * (κ (i.val - j.val)) * (c j) := by
          simp +decide [ Matrix.mulVec, dotProduct, Finset.mul_sum _ _ _, mul_assoc ];
          rfl;
        -- Substitute `κ (i - j) = ∫ z^(i-j) dμ`.
        have h_subst : ∑ i : Fin n, ∑ j : Fin n, (star (c i)) * (κ (i.val - j.val)) * (c j) = ∫ z : Circle, (∑ i : Fin n, ∑ j : Fin n, (star (c i)) * (z ^ (i.val - j.val : ℤ)) * (c j)) ∂μ := by
          rw [ MeasureTheory.integral_finset_sum ];
          · refine' Finset.sum_congr rfl fun i _ => _;
            rw [ MeasureTheory.integral_finset_sum ];
            · simp +decide [ hκ, MeasureTheory.integral_const_mul, MeasureTheory.integral_mul_const ];
              congr! 3;
            · intro j _; apply_rules [ Continuous.integrable_of_hasCompactSupport ] ; continuity;
              exact IsClosed.isCompact ( isClosed_closure ) |> fun h => h.of_isClosed_subset ( isClosed_closure ) ( by aesop_cat );
          · refine' fun i _ => Continuous.integrable_of_hasCompactSupport _ _;
            · continuity;
            · rw [ hasCompactSupport_iff_eventuallyEq ];
              simp +decide [ Filter.EventuallyEq ];
        -- Recognize that the expression inside the integral is the squared norm of the trigonometric polynomial.
        have h_norm_sq : ∀ z : Circle, (∑ i : Fin n, ∑ j : Fin n, (star (c i)) * (z ^ (i.val - j.val : ℤ)) * (c j)) = Complex.normSq (∑ i : Fin n, c i * (z ^ (-i.val : ℤ))) := by
          intro z
          have h_trig_poly : ∀ z : Circle, (∑ i : Fin n, ∑ j : Fin n, (star (c i)) * (z ^ (i.val - j.val : ℤ)) * (c j)) = (∑ i : Fin n, (star (c i)) * (z ^ (i.val : ℤ))) * (∑ j : Fin n, (c j) * (z ^ (-j.val : ℤ))) := by
            intro z; rw [ Finset.sum_mul ] ; rw [ Finset.sum_congr rfl ] ; intros; rw [ Finset.mul_sum ] ; rw [ Finset.sum_congr rfl ] ; intros; ring; norm_num [ zpow_sub₀, zpow_add₀, show z.val ≠ 0 from by aesop ] ; ring;
          simp [h_trig_poly] at *;
          rw [ h_trig_poly, Complex.normSq_apply ] ; norm_num [ Complex.ext_iff ] ; ring;
          norm_num [ Finset.sum_add_distrib, Finset.mul_sum _ _ _, Finset.sum_mul _ _ _, sq ] ; ring ; norm_num;
        exact h_quad_form.trans $ h_subst.trans $ by congr; ext; exact h_norm_sq _;

/-
A complex matrix is positive semidefinite if it is Hermitian and its quadratic form is non-negative for all vectors.
-/
open Complex MeasureTheory Matrix

def ComplexPSD {n : Type*} [Fintype n] (M : Matrix n n ℂ) : Prop :=
  M.IsHermitian ∧ ∀ x : n → ℂ, 0 ≤ (star x ⬝ᵥ M.mulVec x).re

/-
A moment sequence is Hermitian symmetric, meaning `κ(-n)` is the complex conjugate of `κ(n)`.
-/
open Complex MeasureTheory Matrix

theorem moment_seq_is_hermitian {κ : ℤ → ℂ} (h : IsMomentSequence κ) :
    ∀ n : ℤ, κ (-n) = star (κ n) := by
      obtain ⟨ μ, hμ₁, hμ₂ ⟩ := h;
      simp +decide [ hμ₂, Complex.conj_ofReal ];
      intro n; rw [ ← integral_conj ] ; congr; ext; simp +decide [ Complex.inv_def, Complex.normSq_eq_norm_sq ] ;

/-
The Toeplitz matrix of a moment sequence is Hermitian.

theorem toeplitz_is_hermitian {κ : ℤ → ℂ} (h : IsMomentSequence κ) (n : ℕ) :
    (ToeplitzMatrix κ n).IsHermitian := by
      ext i j;
      convert ( moment_seq_is_hermitian h ( j - i ) ) |> Eq.symm using 1;
      unfold ToeplitzMatrix; ring;
