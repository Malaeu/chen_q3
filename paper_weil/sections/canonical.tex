\section{The canonical test and the radical}\label{sec:canonical}

\begin{definition}[Riemann's function, CCM normalisation]\label{def:f0}
Let $h(u)=(\pi^2u^4-\tfrac32\pi u^2)\,e^{-\pi u^2}=\tfrac\pi2u^2(2\pi u^2-3)e^{-\pi u^2}$, the function of
\cite[(7.1)]{CCM-ZST-2025}, and
\begin{equation}\label{eq:Phi}
 \Phi(x)=4\,e^{x/2}\sum_{n\ge1}h(ne^{x}),\qquad A=\|\Phi\|_2,\qquad f_0=\Phi/A .
\end{equation}
\end{definition}

\begin{theorem}\label{thm:FPhi}
$\Phi$ is even, real-analytic, strictly positive, satisfies $|\Phi^{(j)}(x)|\le AM_je^{-\frac{\pi}{2}e^{2|x|}}$ for
$j=0,1$ with $M_0=23.910$, $M_1=325.43$ (\Cref{app:constants}), and
\[
 \widehat\Phi(z)=\Xi(z)\qquad(z\in\C).
\]
\end{theorem}
\begin{proof}
With $\theta(y)=\sum_{n\ge1}e^{-\pi n^2y}$, Riemann's representation
$\xi(s)=\tfrac12-\tfrac{s(1-s)}{2}\int_1^\infty\bigl(y^{s/2}+y^{(1-s)/2}\bigr)\frac{\theta(y)}{y}\,dy$
and two integrations by parts give the classical formula $\Xi(z)=4\int_{\R}\Phi_P(u)\cos(2zu)\,du$, i.e.\
$H_0(z):=\int_0^\infty\Phi_P(u)\cos(zu)\,du=\tfrac18\Xi(z/2)$, with P\'olya's
$\Phi_P(u)=\sum_n(2\pi^2n^4e^{9u}-3\pi n^2e^{5u})e^{-\pi n^2e^{4u}}$ in the normalisation of \cite{RodgersTao2018,Polymath15}.
Substituting $u=x/2$ gives $\Phi(x)=2\Phi_P(x/2)$ term by term, hence $\widehat\Phi(z)=4\int\Phi_P(u)e^{-2izu}du=\Xi(z)$; the same
substitution shows that the de Bruijn--Newman function $H_t(z)=\int_0^\infty e^{tu^2}\Phi_P(u)\cos(zu)\,du$ equals
$\tfrac18\,\widehat{(e^{tx^2/4}\Phi)}(z/2)$. Evenness of $\Phi$ is the functional equation, written as $E(u)=E(1/u)$ for
$E(h)(u)=\sqrt u\sum_nh(nu)$. Positivity: $h(u)<0$ only for $u<\sqrt{3/(2\pi)}=0.691$, so for $x\ge0$ every term $h(ne^x)$ is positive;
for $x<0$ use evenness. The envelope is the $n=1$ term of the Gaussian series with the polynomial prefactor bounded by $M_j$
(\Cref{app:constants}); the remaining terms are smaller by the factor $e^{-\frac\pi2(n^2-1)e^{2|x|}}$.
\end{proof}

\Cref{thm:FPhi} is \cite[Lemma 7.1]{CCM-ZST-2025} in the logarithmic variable: with $E(f)(u)=u^{1/2}\sum_nf(nu)$ as in
\cite[(7.2)]{CCM-ZST-2025}, $\Phi(x)=4E(h)(e^x)$, and CCM attribute the observation that $\Xi$ is the Fourier transform of $E(h)$ to
Riemann \cite{Riemann1859}; the expanded form of $\Phi$ is also eq.~(1) of the preprint \cite{Freedman2026}. We record the factor $4$ and the
argument $e^{x}$ because the multiplicative and additive normalisations are easy to confuse. The normalising constant is
$A=0.565466013092$ (\Cref{app:constants}); $\widehat{f_0}=\Xi/A$ and $\int f_0=\Xi(0)/A=0.8791346724$.

\begin{proposition}[radical; uses \eqref{eq:EF}]\label{prop:radical}
$\B(f_0,v)=0$ for all $v\in\Xf$. More generally, $U_qf_0\in\operatorname{rad}\B$ for every $q\in\R$, and
$f_0*h\in\operatorname{rad}\B$ for every $h\in C_c^\infty$. In particular the radical of $\B$ is infinite-dimensional.
\end{proposition}
\begin{proof}
$f_0$ is not compactly supported, so \eqref{eq:EF} is applied to the cutoffs $g_R=\chi_Rf_0$ of \Cref{lem:budget} and the limit
$R\to\infty$ is taken on both sides. Fix $v\in C_c^\infty$. On the source side, $|\B(g_R,v)-\B(f_0,v)|\le C_{\Xf}\|(1-\chi_R)f_0\|_{\Xf}\|v\|_{\Xf}\to0$
by the bound in \Cref{lem:budget}. On the zero side, $\widehat{g_R}-\widehat{f_0}=\widehat{e_R}$ with $e_R=(1-\chi_R)f_0$, and
$|\widehat{e_R}(z)|\le\|e_R''\|_{L^1(e^{|x|/2}dx)}\,|z|^{-2}$ on the strip $|\Im z|\le\tfrac12$ containing $Z(\Xi)$; since $\widehat v$ is
bounded on that strip ($v$ has compact support) and $\sum_{z\in Z(\Xi)}|z|^{-2}<\infty$, the zero-side sum for $g_R$ converges to
$\sum_z\overline{\Xi(\bar z)}\widehat v(z)/A=0$, because $\widehat{f_0}=\Xi/A$ vanishes at every zero, wherever it lies. Hence
$\B(f_0,v)=0$ for $v\in C_c^\infty$, and by density and the bound $C_{\Xf}$ for all $v\in\Xf$. For the translates: polarising the
translation invariance of every term of \eqref{eq:Q} ($H$, $\D$ and $C_g$ are invariant, and $A_\pm(U_qg)=e^{\pm q/2}A_\pm(g)$ leaves
$A_+\overline{A_-}$ unchanged) gives $\B(U_qg,U_qv)=\B(g,v)$, hence $\B(U_qf_0,v)=\B(f_0,U_{-q}v)=0$. For $f_0*h$ the same cutoff
argument applies with $\widehat{f_0*h}=\Xi\,\widehat h/A$, which also vanishes on $Z(\Xi)$. Finally, finitely many translates of the
positive integrable function $f_0$ are linearly independent (\Cref{lem:indep}), so the radical is infinite-dimensional.
\end{proof}

\begin{remark}[why the class must be enlarged]
Under RH the Weil form is positive \emph{definite} on $C_c^\infty(\R)$ \cite[\S1]{Suzuki2025}, and every finite truncation of Bombieri
is non-degenerate when the zeros are simple \cite[Lemma 10]{BOMBIERI-WEILQF-2000}. \Cref{prop:radical} does not contradict this: $f_0$,
its translates and the $\Xi$-multiples $f_0*h$ are not compactly supported. On the class $\Xf$ the radical contains the closed span of the
$\Xi$-multiples, unconditionally, and Suzuki's Hilbert space \cite{Suzuki2025} is the quotient by the radical; under RH the form is
therefore strictly positive on every nonzero compact test $f_0s$, and \eqref{eq:DOM} below holds with strict inequality.
In the finite picture of \cite[\S5.2]{CCM-ZST-2025} the radical is one-dimensional, $\ker T=\C\xi$, and their trial $k_\lambda$ is the
finite-window version of $f_0$. Numerically, $\Q(f_0)$ computed from \eqref{eq:Q} vanishes to quadrature precision ($10^{-15}$ in our best evaluation), and
$\B(f_0,v)$ for a compactly supported bump $v$ is $-7.5\cdot10^{-16}$; this is the check of the normalisation of \eqref{eq:Q} against
\eqref{eq:EF} mentioned in \Cref{sec:setup}.
\end{remark}
